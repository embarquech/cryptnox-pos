18bool eth_json_result_string(const char *resp, char *out, size_t out_size)
19{
20 bool ok = false;
21 cJSON *root = cJSON_Parse(resp);
22 if (root == NULL) {
23 return false;
24 }
25 const cJSON *result = cJSON_GetObjectItemCaseSensitive(root, "result");
26 if (cJSON_IsString(result) && (result->valuestring != NULL)) {
27 size_t len = strlen(result->valuestring);
28 if ((len + 1U) <= out_size) {
29 ok = CW_Utils::safe_memcpy(
30 reinterpret_cast<uint8_t *>(out), out_size,
31 reinterpret_cast<const uint8_t *>(result->valuestring),
32 len + 1U);
33 }
34 }
35 cJSON_Delete(root);
36 return ok;
37}

39eth_json_receipt_t eth_json_receipt_status(const char *resp)
40{
41 eth_json_receipt_t verdict = ETH_JSON_RECEIPT_ERROR;
42 cJSON *root = cJSON_Parse(resp);
43 if (root == NULL) {
44 return ETH_JSON_RECEIPT_ERROR;
45 }
46 const cJSON *result = cJSON_GetObjectItemCaseSensitive(root, "result");
47 if (cJSON_IsNull(result)) {
48 verdict = ETH_JSON_RECEIPT_PENDING; /* not mined yet */
49 } else if (cJSON_IsObject(result)) {
50 const cJSON *status =
51 cJSON_GetObjectItemCaseSensitive(result, "status");
52 if (cJSON_IsString(status) && (status->valuestring != NULL)) {
53 verdict = (strcmp(status->valuestring, "0x1") == 0)
54 ? ETH_JSON_RECEIPT_SUCCESS
55 : ETH_JSON_RECEIPT_REVERTED;
56 }
57 }
58 cJSON_Delete(root);
59 return verdict;
60}
