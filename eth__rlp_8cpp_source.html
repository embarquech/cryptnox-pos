22/* Maximum scratch buffer for the list content before prepending the header. */
23#define ITEMS_BUF_MAX 320U
24
25/*
26 * Worst-case bytes consumed in items[] by every non-calldata field (F-08):
27 * 5x uint64 (≤9 each = 45) + to (21) + eth_value (≤9) + calldata
28 * header (≤9) + empty access list (1) + v (≤9) + r (≤33) + s (≤33) = 160.
29 * Any calldata longer than what remains would overflow the stack buffer
30 * before the out_max check ever runs, so it is rejected up front.
31 */
32#define ITEMS_FIXED_MAX 160U
33#define CALLDATA_MAX (ITEMS_BUF_MAX - ITEMS_FIXED_MAX)

56static uint8_t be_minimal(uint64_t value, uint8_t out[8])
57{
58 uint8_t tmp[8];
59 uint8_t first = 0U;
60 uint8_t i;
61
62 tmp[0] = static_cast<uint8_t>((value >> 56U) & 0xFFU);
63 tmp[1] = static_cast<uint8_t>((value >> 48U) & 0xFFU);
64 tmp[2] = static_cast<uint8_t>((value >> 40U) & 0xFFU);
65 tmp[3] = static_cast<uint8_t>((value >> 32U) & 0xFFU);
66 tmp[4] = static_cast<uint8_t>((value >> 24U) & 0xFFU);
67 tmp[5] = static_cast<uint8_t>((value >> 16U) & 0xFFU);
68 tmp[6] = static_cast<uint8_t>((value >> 8U) & 0xFFU);
69 tmp[7] = static_cast<uint8_t>(value & 0xFFU);
70
71 while ((first < 7U) && (tmp[first] == 0U)) {
72 first++;
73 }
74 uint8_t count = static_cast<uint8_t>(8U - first);
75 for (i = 0U; i < count; i++) {
76 out[i] = tmp[static_cast<size_t>(first) + i];
77 }
78 return count;
79}

89static size_t rlp_bytes(const uint8_t *data, size_t data_len, uint8_t *out)
90{
91 size_t written = 0U;
92
93 if (data_len == 0U) {
94 out[0] = 0x80U;
95 written = 1U;
96 } else if ((data_len == 1U) && (data[0] < 0x80U)) {
97 out[0] = data[0];
98 written = 1U;
99 } else if (data_len <= 55U) {
100 out[0] = static_cast<uint8_t>(0x80U + data_len);
101 (void)memcpy(out + 1U, data, data_len);
102 written = 1U + data_len;
103 } else {
104 /* Long string: 0xb7 + len(length) bytes, then length, then data. */
105 uint8_t len_bytes[8];
106 uint8_t num_lb = be_minimal(static_cast<uint64_t>(data_len), len_bytes);
107 out[0] = static_cast<uint8_t>(0xB7U + num_lb);
108 (void)memcpy(out + 1U, len_bytes, num_lb);
109 (void)memcpy(out + 1U + num_lb, data, data_len);
110 written = 1U + num_lb + data_len;
111 }
112 return written;
113}

124static size_t rlp_uint64(uint64_t value, uint8_t *out)
125{
126 if (value == 0U) {
127 out[0] = 0x80U;
128 return 1U;
129 }
130 uint8_t be[8];
131 uint8_t be_len = be_minimal(value, be);
132 return rlp_bytes(be, be_len, out);
133}

144static size_t rlp_int256(const uint8_t data[32], uint8_t *out)
145{
146 size_t first = 0U;
147 while ((first < 31U) && (data[first] == 0U)) {
148 first++;
149 }
150 if ((first == 31U) && (data[31] == 0U)) {
151 /* Value is zero */
152 out[0] = 0x80U;
153 return 1U;
154 }
155 return rlp_bytes(data + first, 32U - first, out);
156}

165static size_t rlp_list_header(size_t content_len, uint8_t *out)
166{
167 if (content_len <= 55U) {
168 out[0] = static_cast<uint8_t>(0xC0U + content_len);
169 return 1U;
170 }
171 uint8_t len_bytes[8];
172 uint8_t num_lb = be_minimal(static_cast<uint64_t>(content_len), len_bytes);
173 out[0] = static_cast<uint8_t>(0xF7U + num_lb);
174 (void)memcpy(out + 1U, len_bytes, num_lb);
175 return 1U + num_lb;
176}

192static size_t encode_common_fields(const eth_tx_t *tx, uint8_t *items)
193{
194 if (tx->calldata_len > CALLDATA_MAX) {
195 return 0U;
196 }
197 size_t n = 0U;
198 n += rlp_uint64(tx->chain_id, items + n);
199 n += rlp_uint64(tx->nonce, items + n);
200 n += rlp_uint64(tx->max_priority_fee, items + n);
201 n += rlp_uint64(tx->max_fee, items + n);
202 n += rlp_uint64(tx->gas_limit, items + n);
203 n += rlp_bytes(tx->to, 20U, items + n);
204 n += rlp_uint64(tx->eth_value, items + n);
205 n += rlp_bytes(tx->calldata, tx->calldata_len, items + n);
206 items[n++] = 0xC0U; /* empty access list */
207 return n;
208}

214size_t eth_rlp_encode_unsigned(const eth_tx_t *tx, uint8_t *out, size_t out_max)
215{
216 uint8_t items[ITEMS_BUF_MAX];
217 size_t items_len = encode_common_fields(tx, items);
218 if (items_len == 0U) { return 0U; } /* calldata too large (F-08) */
219
220 uint8_t hdr[10];
221 size_t hdr_len = rlp_list_header(items_len, hdr);
222
223 size_t total = 1U + hdr_len + items_len; /* 0x02 prefix + list */
224 if (total > out_max) { return 0U; }
225
226 out[0] = 0x02U; /* EIP-1559 type prefix */
227 (void)memcpy(out + 1U, hdr, hdr_len);
228 (void)memcpy(out + 1U + hdr_len, items, items_len);
229 return total;
230}

232size_t eth_rlp_encode_signed(const eth_tx_t *tx, uint8_t v,
233 const uint8_t r[32], const uint8_t s[32],
234 uint8_t *out, size_t out_max)
235{
236 uint8_t items[ITEMS_BUF_MAX];
237 size_t items_len = encode_common_fields(tx, items);
238 if (items_len == 0U) { return 0U; } /* calldata too large (F-08) */
239
240 items_len += rlp_uint64(static_cast<uint64_t>(v), items + items_len);
241 items_len += rlp_int256(r, items + items_len);
242 items_len += rlp_int256(s, items + items_len);
243
244 uint8_t hdr[10];
245 size_t hdr_len = rlp_list_header(items_len, hdr);
246
247 size_t total = 1U + hdr_len + items_len;
248 if (total > out_max) { return 0U; }
249
250 out[0] = 0x02U;
251 (void)memcpy(out + 1U, hdr, hdr_len);
252 (void)memcpy(out + 1U + hdr_len, items, items_len);
253 return total;
254}
