63static void wifi_event_handler(void *arg, esp_event_base_t event_base,
64 int32_t event_id, void *event_data)
65{
66 (void)arg;
67 (void)event_data;
68
69 if ((event_base == WIFI_EVENT) && (event_id == WIFI_EVENT_STA_START)) {
70 /* Started — association is initiated explicitly by
71 * net_wifi_connect(), not here, so a start with no credentials
72 * (e.g. before provisioning) doesn't churn through retries. */
73 } else if ((event_base == WIFI_EVENT) &&
74 (event_id == WIFI_EVENT_STA_DISCONNECTED)) {
75 if (s_retry_num < WIFI_MAX_RETRY) {
76 esp_wifi_connect();
77 s_retry_num++;
78 ESP_LOGW(TAG, "WiFi retry %d/%d", s_retry_num, WIFI_MAX_RETRY);
79 } else if (s_wifi_event_group != NULL) {
80 xEventGroupSetBits(s_wifi_event_group, WIFI_FAIL_BIT);
81 }
82 } else if ((event_base == IP_EVENT) && (event_id == IP_EVENT_STA_GOT_IP)) {
83 s_retry_num = 0;
84 if (s_wifi_event_group != NULL) {
85 xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
86 }
87 } else {
88 /* other events ignored */
89 }
90}

96void net_wifi_init(void)
97{
98 if (s_wifi_inited) { return; }
99
100 s_wifi_event_group = xEventGroupCreate();
101 s_retry_num = 0;
102
103 ESP_ERROR_CHECK(esp_netif_init());
104 ESP_ERROR_CHECK(esp_event_loop_create_default());
105 (void)esp_netif_create_default_wifi_sta();
106
107 wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
108 ESP_ERROR_CHECK(esp_wifi_init(&cfg));
109
110 /* Persistent handlers (never unregistered) so init/connect can repeat. */
111 ESP_ERROR_CHECK(esp_event_handler_instance_register(
112 WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL, NULL));
113 ESP_ERROR_CHECK(esp_event_handler_instance_register(
114 IP_EVENT, IP_EVENT_STA_GOT_IP, &wifi_event_handler, NULL, NULL));
115
116 ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
117 ESP_ERROR_CHECK(esp_wifi_start());
118
119 s_wifi_inited = true;
120}

122uint16_t net_wifi_scan(net_wifi_ap_t *out, uint16_t max)
123{
124 net_wifi_init();
125 if ((out == NULL) || (max == 0U)) { return 0U; }
126
127 wifi_scan_config_t scan_cfg;
128 (void)memset(&scan_cfg, 0, sizeof(scan_cfg)); /* all channels, all SSIDs */
129 if (esp_wifi_scan_start(&scan_cfg, true) != ESP_OK) {
130 return 0U;
131 }
132
133 uint16_t num = 0U;
134 (void)esp_wifi_scan_get_ap_num(&num);
135 if (num == 0U) { return 0U; }
136
137 wifi_ap_record_t *recs =
138 static_cast<wifi_ap_record_t *>(malloc(num * sizeof(wifi_ap_record_t)));
139 if (recs == NULL) { return 0U; }
140 (void)esp_wifi_scan_get_ap_records(&num, recs);
141
142 uint16_t count = 0U;
143 for (uint16_t i = 0U; (i < num) && (count < max); i++) {
144 const char *ssid = reinterpret_cast<const char *>(recs[i].ssid);
145 if (ssid[0] == '\0') { continue; } /* hidden network */
146
147 bool dup = false; /* one entry per SSID */
148 for (uint16_t j = 0U; j < count; j++) {
149 if (strcmp(out[j].ssid, ssid) == 0) { dup = true; break; }
150 }
151 if (dup) { continue; }
152
153 (void)strncpy(out[count].ssid, ssid, sizeof(out[count].ssid) - 1U);
154 out[count].ssid[sizeof(out[count].ssid) - 1U] = '\0';
155 out[count].rssi = recs[i].rssi;
156 out[count].open = (recs[i].authmode == WIFI_AUTH_OPEN);
157 count++;
158 }
159
160 free(recs);
161 return count;
162}

164bool net_wifi_connect(const char *ssid, const char *password)
165{
166 net_wifi_init();
167
168 xEventGroupClearBits(s_wifi_event_group, WIFI_CONNECTED_BIT | WIFI_FAIL_BIT);
169 s_retry_num = 0;
170
171 wifi_config_t wifi_cfg;
172 (void)memset(&wifi_cfg, 0, sizeof(wifi_cfg));
173 (void)strncpy(reinterpret_cast<char *>(wifi_cfg.sta.ssid), ssid, sizeof(wifi_cfg.sta.ssid) - 1U);
174 (void)strncpy(reinterpret_cast<char *>(wifi_cfg.sta.password), password, sizeof(wifi_cfg.sta.password) - 1U);
175 /* Open networks have an empty passphrase; otherwise require WPA2+. */
176 wifi_cfg.sta.threshold.authmode =
177 (password[0] == '\0') ? WIFI_AUTH_OPEN : WIFI_AUTH_WPA2_PSK;
178
179 ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wifi_cfg));
180 /* The WiFi driver keeps its own copy — scrub the stack copy of the
181 * credentials immediately (CODING_RULES §1.4). */
182 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(&wifi_cfg),
183 sizeof(wifi_cfg));
184
185 (void)esp_wifi_disconnect(); /* drop any current AP before (re)connecting */
186 (void)esp_wifi_connect();
187
188 ESP_LOGI(TAG, "Connecting to \"%s\"...", ssid);
189
190 EventBits_t bits = xEventGroupWaitBits(s_wifi_event_group,
191 WIFI_CONNECTED_BIT | WIFI_FAIL_BIT,
192 pdFALSE, pdFALSE,
193 pdMS_TO_TICKS(WIFI_TIMEOUT_MS));
194
195 bool connected = ((bits & WIFI_CONNECTED_BIT) != 0U);
196 if (connected) {
197 ESP_LOGI(TAG, "WiFi connected");
198 } else {
199 ESP_LOGE(TAG, "WiFi connect failed");
200 }
201 return connected;
202}

215bool net_time_sync(uint32_t timeout_ms)
216{
217 /* Several reliable servers — phone hotspots often slow or drop NTP (UDP
218 * 123) to pool.ntp.org, so fall back to Google / Cloudflare time. */
219 esp_sntp_config_t config = ESP_NETIF_SNTP_DEFAULT_CONFIG_MULTIPLE(3,
220 ESP_SNTP_SERVER_LIST("time.google.com", "pool.ntp.org", "time.cloudflare.com"));
221 esp_err_t err = esp_netif_sntp_init(&config);
222 if (err != ESP_OK) {
223 ESP_LOGE(TAG, "SNTP init: %s", esp_err_to_name(err));
224 return false;
225 }
226
227 /* Poll across the budget so a slow first response doesn't fail us. */
228 const int attempts = 3;
229 uint32_t per_wait = timeout_ms / static_cast<uint32_t>(attempts);
230 if (per_wait < 2000U) { per_wait = 2000U; }
231 bool synced = false;
232 for (int i = 0; (i < attempts) && !synced; i++) {
233 if (esp_netif_sntp_sync_wait(pdMS_TO_TICKS(per_wait)) == ESP_OK) {
234 synced = true;
235 } else {
236 ESP_LOGW(TAG, "SNTP: no time yet (%d/%d)", i + 1, attempts);
237 }
238 }
239
240 if (!synced) {
241 ESP_LOGE(TAG, "SNTP sync timed out");
242 esp_netif_sntp_deinit();
243 return false;
244 }
245
246 ESP_LOGI(TAG, "System time synced via SNTP");
247 /* Leave SNTP running for periodic background resyncs. */
248 return true;
249}
