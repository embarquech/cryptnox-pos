109 ok = true;

111 ssid[0] = '\0';

185static void admin_derive(const char *code, const uint8_t *salt,
186 uint8_t out[ADMIN_HASH_LEN])
187{
188 uint8_t buf[ADMIN_SALT_LEN + ADMIN_CODE_HASH_MAX];
189 const size_t clen = strnlen(code, ADMIN_CODE_HASH_MAX);
190
191 (void)memcpy(buf, salt, ADMIN_SALT_LEN);
192 (void)memcpy(buf + ADMIN_SALT_LEN, code, clen);
193 keccak256(buf, ADMIN_SALT_LEN + clen, out);
194 CW_Utils::secure_wipe(buf, sizeof(buf));
195}

220bool settings_set_admin_code(const char *code)
221{
222 if (code == NULL) { return false; }
223
224 uint8_t salt[ADMIN_SALT_LEN];
225 esp_fill_random(salt, sizeof(salt));
226
227 uint8_t hash[ADMIN_HASH_LEN];
228 admin_derive(code, salt, hash);
229
230 /* Reported rather than swallowed: the menu — factory reset included — is
231 * unreachable without a stored code, so a silent write failure would leave
232 * a terminal only a USB erase can rescue. */
233 bool ok = false;
234 nvs_handle_t h;
235 if (nvs_open(NS_SETTINGS, NVS_READWRITE, &h) == ESP_OK) {
236 ok = (nvs_set_blob(h, K_ADMIN_SALT, salt, sizeof(salt)) == ESP_OK) &&
237 (nvs_set_blob(h, K_ADMIN_HASH, hash, sizeof(hash)) == ESP_OK) &&
238 (nvs_set_u8(h, K_ADMIN_FAILS, 0U) == ESP_OK) &&
239 (nvs_commit(h) == ESP_OK);
240 nvs_close(h);
241 ESP_LOGI(TAG, "admin code set: %s", ok ? "ok" : "FAILED");
242 } else {
243 ESP_LOGW(TAG, "admin code: nvs_open failed");
244 }
245 CW_Utils::secure_wipe(hash, sizeof(hash));
246 return ok;
247}

249bool settings_check_admin_code(const char *code)
250{
251 if (code == NULL) { return false; }
252
253 uint8_t salt[ADMIN_SALT_LEN];
254 uint8_t stored[ADMIN_HASH_LEN];
255 bool have = false;
256
257 nvs_handle_t h;
258 if (nvs_open(NS_SETTINGS, NVS_READONLY, &h) == ESP_OK) {
259 size_t ls = sizeof(salt);
260 size_t lh = sizeof(stored);
261 have = (nvs_get_blob(h, K_ADMIN_SALT, salt, &ls) == ESP_OK) &&
262 (nvs_get_blob(h, K_ADMIN_HASH, stored, &lh) == ESP_OK) &&
263 (ls == ADMIN_SALT_LEN) && (lh == ADMIN_HASH_LEN);
264 nvs_close(h);
265 }
266 if (!have) { return false; } /* no code stored — nothing to match */
267
268 uint8_t calc[ADMIN_HASH_LEN];
269 admin_derive(code, salt, calc);
270 const bool ok = CW_Utils::secure_compare(calc, stored, ADMIN_HASH_LEN);
271 CW_Utils::secure_wipe(calc, sizeof(calc));
272
273 if (ok) {
274 if (settings_admin_fail_count() != 0U) { admin_set_fails(0U); }
275 } else {
276 const uint8_t n = settings_admin_fail_count();
277 admin_set_fails((n < 255U) ? (uint8_t)(n + 1U) : 255U);
278 ESP_LOGW(TAG, "admin unlock failed (%u consecutive)", (unsigned)(n + 1U));
279 }
280 return ok;
281}

295{

297 if (nvs_open(NS_SETTINGS, NVS_READWRITE, &h) == ESP_OK) {
298 /* Drops brightness, Wi-Fi creds and the admin code, so a reset terminal
