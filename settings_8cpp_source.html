36uint8_t settings_get_brightness(void)
37{
38 uint8_t val = DEFAULT_BRIGHTNESS;
39 nvs_handle_t h;
40 if (nvs_open(NS_SETTINGS, NVS_READONLY, &h) == ESP_OK) {
41 uint8_t stored;
42 if (nvs_get_u8(h, K_BRIGHTNESS, &stored) == ESP_OK) {
43 val = stored;
44 }
45 nvs_close(h);
46 }

84bool settings_has_wifi(void)
85{

88 if (nvs_open(NS_SETTINGS, NVS_READONLY, &h) == ESP_OK) {
89 char ssid[33] = {0};
90 size_t len = sizeof(ssid);

93 nvs_close(h);

98bool settings_get_wifi(char *ssid, size_t ssid_n, char *pass, size_t pass_n)
99{
100 if ((ssid == NULL) || (pass == NULL) || (ssid_n == 0U) || (pass_n == 0U)) {
101 return false;
102 }
103 ssid[0] = '\0';
104 pass[0] = '\0';
105
106 bool ok = false;
107 nvs_handle_t h;
108 if (nvs_open(NS_SETTINGS, NVS_READONLY, &h) == ESP_OK) {
109 size_t ls = ssid_n;
110 size_t lp = pass_n;
111 if ((nvs_get_str(h, K_WIFI_SSID, ssid, &ls) == ESP_OK) &&
112 (nvs_get_str(h, K_WIFI_PASS, pass, &lp) == ESP_OK) &&
113 (ssid[0] != '\0')) {
114 ok = true;
115 } else {
116 ssid[0] = '\0';
117 pass[0] = '\0';
118 }
119 nvs_close(h);
120 }
121 return ok;
122}
