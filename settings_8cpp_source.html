35uint8_t settings_get_brightness(void)
36{
37 uint8_t val = DEFAULT_BRIGHTNESS;
38 nvs_handle_t h;
39 if (nvs_open(NS_SETTINGS, NVS_READONLY, &h) == ESP_OK) {
40 uint8_t stored;
41 if (nvs_get_u8(h, K_BRIGHTNESS, &stored) == ESP_OK) {
42 val = stored;
43 }
44 nvs_close(h);
45 }
46 return val;
47}

62bool settings_has_wifi(void)

67 char ssid[33] = {0};

76bool settings_get_wifi(char *ssid, size_t ssid_n, char *pass, size_t pass_n)
77{
78 if ((ssid == NULL) || (pass == NULL) || (ssid_n == 0U) || (pass_n == 0U)) {
79 return false;
80 }
81 ssid[0] = '\0';
82 pass[0] = '\0';
83
84 bool ok = false;
85 nvs_handle_t h;
86 if (nvs_open(NS_SETTINGS, NVS_READONLY, &h) == ESP_OK) {
87 size_t ls = ssid_n;
88 size_t lp = pass_n;
89 if ((nvs_get_str(h, K_WIFI_SSID, ssid, &ls) == ESP_OK) &&
90 (nvs_get_str(h, K_WIFI_PASS, pass, &lp) == ESP_OK) &&
91 (ssid[0] != '\0')) {
92 ok = true;
93 } else {
94 ssid[0] = '\0';
95 pass[0] = '\0';
96 }
97 nvs_close(h);
98 }
99 return ok;
100}
