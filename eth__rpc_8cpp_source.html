80static bool do_post(const char *body, char *resp_buf, size_t resp_buf_size)
81{
82 bool success = false;
83
84 bool use_auth = ((s_project_id != NULL) && (s_project_id[0] != '\0') &&
85 (s_api_secret != NULL) && (s_api_secret[0] != '\0'));
86
87 esp_http_client_config_t cfg;
88 (void)memset(&cfg, 0, sizeof(cfg));
89 cfg.url = s_rpc_url;
90 cfg.method = HTTP_METHOD_POST;
91 cfg.timeout_ms = 15000;
92 /* if a cert was pinned via eth_rpc_set_ca_cert(), trust ONLY it —
93 * otherwise any of the ~150 CAs in the Mozilla bundle could MITM the RPC. */
94 if (s_ca_cert != NULL) {
95 cfg.cert_pem = s_ca_cert;
96 } else {
97 cfg.crt_bundle_attach = esp_crt_bundle_attach;
98 }
99 if (use_auth) {
100 cfg.username = s_project_id;
101 cfg.password = s_api_secret;
102 cfg.auth_type = HTTP_AUTH_TYPE_BASIC;
103 }
104
105 esp_http_client_handle_t client = esp_http_client_init(&cfg);
106 if (client == NULL) {
107 ESP_LOGE(TAG, "HTTP client init failed");
108 return false;
109 }
110
111 (void)esp_http_client_set_header(client, "Content-Type", "application/json");
112
113 int body_len = static_cast<int>(strlen(body));
114 esp_err_t err = esp_http_client_open(client, body_len);
115 if (err != ESP_OK) {
116 ESP_LOGE(TAG, "HTTP open: %s", esp_err_to_name(err));
117 goto cleanup;
118 }
119
120 if (esp_http_client_write(client, body, body_len) != body_len) {
121 ESP_LOGE(TAG, "HTTP write incomplete");
122 goto cleanup;
123 }
124
125 {
126 int64_t content_length = esp_http_client_fetch_headers(client);
127 (void)content_length; /* may be -1 for chunked; we read until EOF */
128
129 int total = 0;
130 int read;
131 do {
132 int space = static_cast<int>(resp_buf_size - 1U) - total;
133 if (space <= 0) { break; }
134 read = esp_http_client_read(client, resp_buf + total, space);
135 if (read > 0) { total += read; }
136 } while (read > 0);
137
138 resp_buf[total] = '\0';
139
140 /* a 4xx/5xx body that happens to contain "result" must not
141 * be mistaken for a successful JSON-RPC response. */
142 int status = esp_http_client_get_status_code(client);
143 if (status != 200) {
144 ESP_LOGE(TAG, "HTTP status %d: %.*s", status, RESP_LOG_MAX, resp_buf);
145 }
146 success = ((total > 0) && (status == 200));
147 }
148
149cleanup:
150 esp_http_client_close(client);
151 esp_http_client_cleanup(client);
152 return success;
153}

165static char hex_nibble(uint8_t n)
166{
167 return (n < 10U) ? static_cast<char>('0' + n)
168 : static_cast<char>('a' + n - 10U);
169}

178static void bytes_to_hex(const uint8_t *data, size_t len, char *out)
179{
180 size_t i;
181 for (i = 0U; i < len; i++) {
182 out[i * HEX_PER_BYTE] = hex_nibble((data[i] >> 4U) & 0x0FU);
183 out[i * HEX_PER_BYTE + 1U] = hex_nibble(data[i] & 0x0FU);
184 }
185}

211bool eth_rpc_get_nonce(uint64_t *nonce_out)
212{
213 char body[256];
214 (void)snprintf(body, sizeof(body),
215 "{\"jsonrpc\":\"2.0\",\"method\":\"eth_getTransactionCount\","
216 "\"params\":[\"%s\",\"pending\"],\"id\":1}",
217 s_from_addr);
218
219 char resp[RESP_BUF_SIZE];
220 if (!do_post(body, resp, sizeof(resp))) {
221 return false;
222 }
223
224 char result[RESULT_STR_MAX];
225 if (!eth_json_result_string(resp, result, sizeof(result))) {
226 ESP_LOGE(TAG, "nonce: no result in: %.*s", RESP_LOG_MAX, resp);
227 return false;
228 }
229 if (strncmp(result, "0x", 2U) != 0) {
230 ESP_LOGE(TAG, "nonce: result not hex: %.*s", RESP_LOG_MAX, result);
231 return false;
232 }
233
234 char *end = NULL;
235 uint64_t nonce = strtoull(result + 2, &end, 16);
236 if ((end == (result + 2)) || (*end != '\0')) {
237 ESP_LOGE(TAG, "nonce: malformed hex: %.*s", RESP_LOG_MAX, result);
238 return false;
239 }
240 /* strtoull saturates silently — reject absurd values outright. */
241 if (nonce > NONCE_MAX) {
242 ESP_LOGE(TAG, "nonce: out of range: %" PRIu64, nonce);
243 return false;
244 }
245
246 *nonce_out = nonce;
247 ESP_LOGI(TAG, "Nonce: %" PRIu64, nonce);
248 return true;
249}

251eth_rpc_parity_result_t eth_rpc_ecrecover_parity(const uint8_t hash[32],
252 const uint8_t r[32],
253 const uint8_t s[32],
254 uint8_t *v_out)
255{
256 /* Track whether at least one eth_call produced a usable recovered
257 * address, so the caller can distinguish "RPC down" from "address
258 * mismatch". */
259 bool got_recovered = false;
260
261 /* ecrecover precompile input: hash(32) || v_uint256(32) || r(32) || s(32) */
262 uint8_t input[128];
263 (void)memset(input, 0, sizeof(input));
264 (void)CW_Utils::safe_memcpy(input, sizeof(input), hash, 32U);
265 /* v occupies the last byte of the second 32-byte slot (index 63) */
266 (void)CW_Utils::safe_memcpy(input + 64U, sizeof(input) - 64U, r, 32U);
267 (void)CW_Utils::safe_memcpy(input + 96U, sizeof(input) - 96U, s, 32U);
268
269 /* Hex-encode the 128-byte input */
270 char input_hex[257];
271 bytes_to_hex(input, sizeof(input), input_hex);
272 input_hex[256] = '\0';
273
274 /* from_addr without "0x" prefix for comparison */
275 const char *from_hex = s_from_addr;
276 if ((from_hex[0] == '0') && ((from_hex[1] == 'x') || (from_hex[1] == 'X'))) {
277 from_hex += 2;
278 }
279
280 uint8_t v_raw;
281 for (v_raw = 0U; v_raw < 2U; v_raw++) {
282 /* Set v byte (27 or 28) in slot [32..63] last byte */
283 input_hex[63U * HEX_PER_BYTE] = hex_nibble(((27U + v_raw) >> 4U) & 0x0FU);
284 input_hex[63U * HEX_PER_BYTE + 1U] = hex_nibble((27U + v_raw) & 0x0FU);
285
286 char body[600];
287 (void)snprintf(body, sizeof(body),
288 "{\"jsonrpc\":\"2.0\",\"method\":\"eth_call\","
289 "\"params\":[{\"to\":"
290 "\"0x0000000000000000000000000000000000000001\","
291 "\"data\":\"0x%s\"},\"latest\"],\"id\":3}",
292 input_hex);
293
294 char resp[RESP_BUF_SIZE];
295 if (!do_post(body, resp, sizeof(resp))) {
296 continue;
297 }
298
299 /* Expected result: "0x" + 64 hex chars (32-byte ABI-encoded address).
300 * The address occupies the last 40 hex chars (bytes 12-31). */
301 char result[RESULT_STR_MAX];
302 if (!eth_json_result_string(resp, result, sizeof(result))) {
303 ESP_LOGW(TAG, "ecrecover v=%u: no result in: %.*s",
304 v_raw, RESP_LOG_MAX, resp);
305 continue;
306 }
307 if ((strncmp(result, "0x", 2U) != 0) || (strlen(result) != 66U)) {
308 /* Empty/short result — address not recovered (try other v) */
309 continue;
310 }
311 got_recovered = true;
312
313 /* ABI address: 24 hex chars of zeros + 40 hex chars of address */
314 const char *recovered_hex = result + 2U + 24U;
315
316 if (strncasecmp(recovered_hex, from_hex, 40U) == 0) {
317 ESP_LOGI(TAG, "v=%u matched ecrecover", v_raw);
318 *v_out = v_raw;
319 return ETH_RPC_PARITY_OK;
320 }
321 }
322
323 /* no silent v=0 fallback — broadcasting with a wrong parity just
324 * produces an invalid signature and an opaque failure downstream. */
325 if (got_recovered) {
326 ESP_LOGE(TAG, "ecrecover: neither parity matches from_addr "
327 "(ADDR_FROM / card mismatch?)");
328 return ETH_RPC_PARITY_MISMATCH;
329 }
330 ESP_LOGE(TAG, "ecrecover: no usable RPC response for either parity");
331 return ETH_RPC_PARITY_RPC_ERROR;
332}

334bool eth_rpc_send_raw_tx(const uint8_t *tx, size_t tx_len,
335 char *tx_hash_out, size_t tx_hash_max)
336{
337 /* "0x" + 2 hex chars per byte + NUL */
338 size_t hex_str_size = 2U + tx_len * HEX_PER_BYTE + 1U;
339 char *tx_hex = static_cast<char *>(malloc(hex_str_size));
340 if (tx_hex == NULL) { return false; }
341
342 tx_hex[0] = '0';
343 tx_hex[1] = 'x';
344 bytes_to_hex(tx, tx_len, tx_hex + 2U);
345 tx_hex[hex_str_size - 1U] = '\0';
346
347 /* JSON body */
348 size_t body_size = hex_str_size + 128U;
349 char *body = static_cast<char *>(malloc(body_size));
350 if (body == NULL) { free(tx_hex); return false; }
351
352 (void)snprintf(body, body_size,
353 "{\"jsonrpc\":\"2.0\",\"method\":\"eth_sendRawTransaction\","
354 "\"params\":[\"%s\"],\"id\":2}",
355 tx_hex);
356 free(tx_hex);
357
358 char resp[RESP_BUF_SIZE];
359 bool ok = do_post(body, resp, sizeof(resp));
360 free(body);
361
362 if (!ok) { return false; }
363
364 /* Extract the "result" string (the tx hash) with a real JSON parse, so
365 * a JSON-RPC error object is reported as a failure. */
366 char result[RESULT_STR_MAX];
367 if (!eth_json_result_string(resp, result, sizeof(result))) {
368 ESP_LOGE(TAG, "send_raw_tx: no result in: %.*s", RESP_LOG_MAX, resp);
369 return false;
370 }
371 if (strncmp(result, "0x", 2U) != 0) {
372 ESP_LOGE(TAG, "send_raw_tx: result not a hash: %.*s",
373 RESP_LOG_MAX, result);
374 return false;
375 }
376
377 size_t hash_len = strlen(result);
378 if ((hash_len + 1U) > tx_hash_max) { return false; }
379
380 (void)CW_Utils::safe_memcpy(reinterpret_cast<uint8_t *>(tx_hash_out),
381 tx_hash_max,
382 reinterpret_cast<const uint8_t *>(result),
383 hash_len);
384 tx_hash_out[hash_len] = '\0';
385 ESP_LOGI(TAG, "Tx hash: %s", tx_hash_out);
386 return true;
387}

389eth_rpc_receipt_result_t eth_rpc_get_tx_receipt(const char *tx_hash)
390{
391 char body[160];
392 (void)snprintf(body, sizeof(body),
393 "{\"jsonrpc\":\"2.0\",\"method\":\"eth_getTransactionReceipt\","
394 "\"params\":[\"%s\"],\"id\":3}",
395 tx_hash);
396
397 /* Receipts are large (the logsBloom field alone is 512 hex chars, plus
398 * the ERC-20 Transfer log) — use a dedicated heap buffer, a truncated
399 * body would fail the JSON parse. */
400 const size_t resp_size = 4096U;
401 char *resp = static_cast<char *>(malloc(resp_size));
402 if (resp == NULL) { return ETH_RPC_RECEIPT_RPC_ERROR; }
403
404 eth_rpc_receipt_result_t verdict = ETH_RPC_RECEIPT_RPC_ERROR;
405 if (do_post(body, resp, resp_size)) {
406 switch (eth_json_receipt_status(resp)) {
407 case ETH_JSON_RECEIPT_PENDING: verdict = ETH_RPC_RECEIPT_PENDING; break;
408 case ETH_JSON_RECEIPT_SUCCESS: verdict = ETH_RPC_RECEIPT_SUCCESS; break;
409 case ETH_JSON_RECEIPT_REVERTED: verdict = ETH_RPC_RECEIPT_REVERTED; break;
410 case ETH_JSON_RECEIPT_ERROR: /* fall through */
411 default: verdict = ETH_RPC_RECEIPT_RPC_ERROR; break;
412 }
413 }
414 free(resp);
415 return verdict;
416}
