29bool eth_addr_parse(const char *hex, uint8_t out[20])
30{
31 const char *p = hex;
32 if ((p[0] == '0') && ((p[1] == 'x') || (p[1] == 'X'))) {
33 p += 2;
34 }
35 if (strlen(p) != 40U) {
36 return false;
37 }
38 (void)memset(out, 0, 20U);
39 size_t i;
40 for (i = 0U; i < 20U; i++) {
41 int hi = hex_nibble_val(p[i * 2U]);
42 int lo = hex_nibble_val(p[(i * 2U) + 1U]);
43 if ((hi < 0) || (lo < 0)) {
44 return false;
45 }
46 out[i] = static_cast<uint8_t>((static_cast<uint8_t>(hi) << 4U) |
47 static_cast<uint8_t>(lo));
48 }
49 return true;
