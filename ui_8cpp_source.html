116static bool touch_to_screen(int16_t *sx, int16_t *sy) {
117 if (!touch.tirqTouched() || !touch.touched()) {
118 return false;
119 }
120 TS_Point p = touch.getPoint();
121 int16_t mx = map(p.x, 200, 3800, 0, SCR_W);
122 int16_t my = map(p.y, 200, 3800, 0, SCR_H);
123 if (mx < 0) { mx = 0; }
124 if (mx >= SCR_W) { mx = SCR_W - 1; }
125 if (my < 0) { my = 0; }
126 if (my >= SCR_H) { my = SCR_H - 1; }
127 *sx = mx;
128 *sy = my;
129 return true;
130}

132/* Swallow taps carried over from the screen we just left: a press is ignored
133 * until this tick (ms) AND until the finger has been released at least once.
134 * Set on every screen (re)build in render_requested_screen(). */
135static uint32_t s_input_block_until = 0;

138static void indev_read(lv_indev_drv_t *drv, lv_indev_data_t *data) {
139 (void)drv;
140 int16_t x, y;
141 bool pressed = touch_to_screen(&x, &y);
142
143 /* After a screen change, ignore a lingering or reflexive tap from the old
144 * screen (e.g. cancelling the tx right after validating the PIN). */
145 if (lv_tick_get() < s_input_block_until || s_wait_release) {
146 if (!pressed) { s_wait_release = false; } /* finger lifted — re-arm */
147 data->state = LV_INDEV_STATE_REL;
148 return;
149 }
150
151 if (pressed) {
152 data->state = LV_INDEV_STATE_PR;
153 data->point.x = x;
154 data->point.y = y;
155 } else {
156 data->state = LV_INDEV_STATE_REL;
157 }
158}

229/* Map the ambient-light reading to a backlight %. Brighter room -> brighter
230 * screen; if it's backwards on your unit, swap LDR_PCT_MIN/MAX. */
231static uint8_t ldr_brightness_pct(void) {
232 int raw = 0;
233 if ((s_adc == NULL) ||
234 (adc_oneshot_read(s_adc, LDR_ADC_CHANNEL, &raw) != ESP_OK)) {
235 return s_brightness; /* fall back to the manual level */
236 }
237 if (raw < LDR_RAW_MIN) { raw = LDR_RAW_MIN; }
238 if (raw > LDR_RAW_MAX) { raw = LDR_RAW_MAX; }
239 int pct = LDR_PCT_MIN + (raw - LDR_RAW_MIN) * (LDR_PCT_MAX - LDR_PCT_MIN)
240 / (LDR_RAW_MAX - LDR_RAW_MIN);
241 return static_cast<uint8_t>(pct);
242}

345/* Keypad on the amount screen — cents entry: each digit shifts in
346 * from the right (1 -> 0.01, 12 -> 0.12, 1250 -> 12.50). No decimal point. */
347static void amount_kbd_cb(lv_event_t *e) {
348 lv_obj_t *bm = lv_event_get_target(e);
349 const char *txt = lv_btnmatrix_get_btn_text(bm, lv_btnmatrix_get_selected_btn(bm));
350 if (txt == NULL) { return; }
351
352 if (strcmp(txt, LV_SYMBOL_BACKSPACE) == 0) {
353 s_amount_cents /= 10ULL;
354 } else if (strcmp(txt, "00") == 0) {
355 uint64_t n = s_amount_cents * 100ULL;
356 s_amount_cents = (n > AMOUNT_CENTS_MAX) ? AMOUNT_CENTS_MAX : n;
357 } else if ((txt[0] >= '0') && (txt[0] <= '9') && (txt[1] == '\0')) {
358 uint64_t n = (s_amount_cents * 10ULL) + static_cast<uint64_t>(txt[0] - '0');
359 if (n <= AMOUNT_CENTS_MAX) { s_amount_cents = n; }
360 }
361 amount_update_display();
362}

364/* Runs on the UI task (inside lv_timer_handler), so touching shared state and
365 * invoking s_cb (which only posts to a queue) is safe here. */
366static void btn_event_cb(lv_event_t *e) {
367 BtnAction act = static_cast<BtnAction>(
368 reinterpret_cast<intptr_t>(lv_event_get_user_data(e)));
369
370 switch (act) {
371 case ACT_CONFIRM:
372 if (s_cb != NULL && s_amount_units > 0ULL) {
373 s_cb(UI_EVENT_AMOUNT_CONFIRMED, s_amount_units);
374 }
375 break;
376 case ACT_CANCEL:
377 if (s_cb != NULL) { s_cb(UI_EVENT_CONFIRM_CANCEL, 0); }
378 break;
379 case ACT_SEND:
380 /* The PIN is no longer hard-coded: collect it on the keypad screen
381 * before signing. */
382 request_screen(UI_SCREEN_PIN);
383 break;
384 case ACT_PIN_CANCEL:
385 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(s_pin), sizeof(s_pin));
386 s_pin_len = 0;
387 request_screen(UI_SCREEN_CONFIRM);
388 break;
389 case ACT_NEW:
390 if (s_cb != NULL) { s_cb(UI_EVENT_TX_RETRY, 0); }
391 break;
392 case ACT_SETTINGS:
393 s_settings_return = s_req_screen; /* remember where we came from */
394 request_screen(UI_SCREEN_SETTINGS);
395 break;
396 case ACT_CLOSE:
397 settings_persist();
398 request_screen(s_settings_return);
399 break;
400 case ACT_WIFI:
401 settings_persist();
402 strncpy(s_wifi_msg, "Scanning...", sizeof(s_wifi_msg) - 1);
403 s_wifi_msg[sizeof(s_wifi_msg) - 1] = '\0';
404 request_screen(UI_SCREEN_WIFI_CONNECTING);
405 if (s_cb != NULL) { s_cb(UI_EVENT_WIFI_SCAN, 0); }
406 break;
407 case ACT_WIFI_CANCEL:
408 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(s_wifi_pass),
409 sizeof(s_wifi_pass));
410 request_screen(UI_SCREEN_AMOUNT);
411 break;
412 case ACT_RESET:
413 open_reset_confirm(); /* ask before wiping */
414 break;
415 case ACT_RESET_CONFIRM:
416 /* Wipe stored settings (brightness, auto, Wi-Fi) and reboot — the
417 * device comes back up into first-run Wi-Fi setup. */
418 settings_factory_reset();
419 esp_restart();
420 break;
421 case ACT_RESET_CANCEL:
422 close_reset_confirm();
423 break;
424 }
425}

574 lv_label_set_text(s_maxfee_lbl, b);

582/* user_data encodes which fee and direction: 0 max-, 1 max+, 2 prio-, 3 prio+. */
583static void fee_step_cb(lv_event_t *e) {
584 intptr_t code = reinterpret_cast<intptr_t>(lv_event_get_user_data(e));
585 uint32_t *v = (code < 2) ? &s_maxfee_gwei : &s_prio_gwei;
586 bool inc = (code & 1) != 0;
587
588 if (inc) {
589 *v = (*v + FEE_STEP_GWEI > FEE_MAX_GWEI) ? FEE_MAX_GWEI : *v + FEE_STEP_GWEI;
590 } else {
591 *v = (*v < FEE_MIN_GWEI + FEE_STEP_GWEI) ? FEE_MIN_GWEI : *v - FEE_STEP_GWEI;
592 }
593 /* The tip can never exceed the cap. */
594 if (s_prio_gwei > s_maxfee_gwei) { s_prio_gwei = s_maxfee_gwei; }
595 fee_update_labels();
596}

1308/******************************************************************
1309 * 9. UI task — owns LVGL init and the handler loop
1310 ******************************************************************/
1311static void ui_task(void *arg) {
1312 (void)arg;
1313
1314 lv_init();
1315
1316 tft.init();
1317 tft.setRotation(0); /* portrait, 240x320 */
1318 tft.invertDisplay(true); /* CYD ILI9341 panel renders inverted otherwise */
1319
1320 /* CYD "milky gamma" fix: the 1-USB ILI9341_2 panels ship with a gamma
1321 * curve that crushes smooth gradients into visible bands — solid colours
1322 * and text look fine, but anti-aliased greys (e.g. the logo edges) come
1323 * out blocky. Re-select a built-in gamma curve via GAMMASET (0x26) to get
1324 * a clean ramp. See TFT_eSPI discussion #3018.
1325 */
1326 tft.writecommand(0x26); /* GAMMASET */
1327 tft.writedata(0x02);
1328 delay(120);
1329 tft.writecommand(0x26);
1330 tft.writedata(0x01);
1331
1332 tft.fillScreen(TFT_BLACK);
1333
1334 touchSPI.begin(T_CLK, T_MISO, T_MOSI, T_CS);
1335 touch.begin(touchSPI);
1336 touch.setRotation(0); /* match the panel orientation */
1337
1338 /* Take over the backlight pin with LEDC PWM (after tft.init has touched
1339 * it) so brightness is dimmable from the settings menu. Restore the saved
1340 * level from NVS (defaults to 80% if never set). */
1341 s_brightness = settings_get_brightness();
1342 s_auto_brightness = settings_get_auto_brightness();
1343 backlight_init(s_brightness);
1344 ldr_init();
1345
1346 lv_disp_draw_buf_init(&s_draw_buf, s_buf, NULL, SCR_W * 40);
1347 lv_disp_drv_init(&s_disp_drv);
1348 s_disp_drv.hor_res = SCR_W;
