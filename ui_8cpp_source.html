121static bool touch_to_screen(int16_t *sx, int16_t *sy) {
122 if (!touch.tirqTouched() || !touch.touched()) {
123 return false;
124 }
125 TS_Point p = touch.getPoint();
126 int16_t mx = map(p.x, 200, 3800, 0, SCR_W);
127 int16_t my = map(p.y, 200, 3800, 0, SCR_H);
128 if (mx < 0) { mx = 0; }
129 if (mx >= SCR_W) { mx = SCR_W - 1; }
130 if (my < 0) { my = 0; }
131 if (my >= SCR_H) { my = SCR_H - 1; }
132 *sx = mx;
133 *sy = my;
134 return true;
135}

137/* Swallow taps carried over from the screen we just left: a press is ignored
138 * until this tick (ms) AND until the finger has been released at least once.
139 * Set on every screen (re)build in render_requested_screen(). */
140static uint32_t s_input_block_until = 0;
141static bool s_wait_release = false;

143static void indev_read(lv_indev_drv_t *drv, lv_indev_data_t *data) {
144 (void)drv;
145 int16_t x, y;
146 bool pressed = touch_to_screen(&x, &y);
147
148 /* After a screen change, ignore a lingering or reflexive tap from the old
149 * screen (e.g. cancelling the tx right after validating the PIN). */
150 if (lv_tick_get() < s_input_block_until || s_wait_release) {
151 if (!pressed) { s_wait_release = false; } /* finger lifted — re-arm */
152 data->state = LV_INDEV_STATE_REL;
153 return;
154 }
155
156 if (pressed) {
157 data->state = LV_INDEV_STATE_PR;
158 data->point.x = x;
159 data->point.y = y;
160 } else {
161 data->state = LV_INDEV_STATE_REL;
162 }
163}

362/* Keypad on the amount screen — cents entry: each digit shifts in
363 * from the right (1 -> 0.01, 12 -> 0.12, 1250 -> 12.50). No decimal point. */
364static void amount_kbd_cb(lv_event_t *e) {
365 lv_obj_t *bm = lv_event_get_target(e);
366 const char *txt = lv_btnmatrix_get_btn_text(bm, lv_btnmatrix_get_selected_btn(bm));
367 if (txt == NULL) { return; }
368
369 if (strcmp(txt, LV_SYMBOL_BACKSPACE) == 0) {
370 s_amount_cents /= 10ULL;
371 } else if (strcmp(txt, "00") == 0) {
372 uint64_t n = s_amount_cents * 100ULL;
373 s_amount_cents = (n > AMOUNT_CENTS_MAX) ? AMOUNT_CENTS_MAX : n;
374 } else if ((txt[0] >= '0') && (txt[0] <= '9') && (txt[1] == '\0')) {
375 uint64_t n = (s_amount_cents * 10ULL) + static_cast<uint64_t>(txt[0] - '0');
376 if (n <= AMOUNT_CENTS_MAX) { s_amount_cents = n; }
377 }
378 amount_update_display();
379}

381/* Runs on the UI task (inside lv_timer_handler), so touching shared state and
382 * invoking s_cb (which only posts to a queue) is safe here. */
383static void btn_event_cb(lv_event_t *e) {
384 BtnAction act = static_cast<BtnAction>(
385 reinterpret_cast<intptr_t>(lv_event_get_user_data(e)));
386
387 switch (act) {
388 case ACT_CONFIRM:
389 if (s_cb != NULL && s_amount_units > 0ULL) {
390 s_cb(UI_EVENT_AMOUNT_CONFIRMED, s_amount_units);
391 }
392 break;
393 case ACT_CANCEL:
394 if (s_cb != NULL) { s_cb(UI_EVENT_CONFIRM_CANCEL, 0); }
395 break;
396 case ACT_SEND:
397 /* The PIN is no longer hard-coded: collect it on the keypad screen
398 * before signing. */
399 request_screen(UI_SCREEN_PIN);
400 break;
401 case ACT_PIN_CANCEL:
402 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(s_pin), sizeof(s_pin));
403 s_pin_len = 0;
404 request_screen(UI_SCREEN_CONFIRM);
405 break;
406 case ACT_NEW:
407 if (s_cb != NULL) { s_cb(UI_EVENT_TX_RETRY, 0); }
408 break;
409 case ACT_SETTINGS:
410 s_settings_return = s_req_screen; /* remember where we came from */
411 /* One lock for the whole menu: Wi-Fi, fee caps and the factory reset
412 * are all merchant operations, and the reset in particular must not
413 * be one tap away from a customer left alone with the terminal.
414 *
415 * No code stored means first-run setup has not finished, so the tap
416 * is ignored rather than let through. main creates the code before
417 * anything else, so this state is never reachable for long — but it
418 * WAS reachable, by backing out of the first-run Wi-Fi picker onto
419 * the amount screen while main was still waiting. */
420 if (!settings_has_admin_code()) { break; }
421 s_admin_confirming = false;
422 s_admin_note[0] = '\0';
423 /* Re-arm the wait from the persisted attempt count. The wait itself
424 * has to live in RAM — persisting a deadline would need a trustworthy
425 * absolute clock, and the wall clock is exactly what an attacker on
426 * the network can move — so a power cycle used to clear it and bring
427 * the cost of one guess down to a single reboot. Deriving it here
428 * instead makes the escalation survive reboots, at no extra write. */
429 s_admin_lock_ms = admin_penalty_ms(settings_admin_fail_count());
430 s_admin_lock_start = lv_tick_get();
431 request_screen(UI_SCREEN_ADMIN_UNLOCK);
432 break;
433 case ACT_WELCOME_OK:
434 /* main answers by showing the code screen straight away, so there is
435 * nothing to fill here. Guarded all the same: request_screen only
436 * takes effect on the UI task's next pass, so a double tap could
437 * otherwise emit twice. */
438 if (!s_welcome_sent) {
439 s_welcome_sent = true;
440 if (s_cb != NULL) { s_cb(UI_EVENT_WELCOME_DONE, 0); }
441 }
442 break;
443 case ACT_ADMIN_CANCEL:
444 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(s_admin_first),
445 sizeof(s_admin_first));
446 request_screen(UI_SCREEN_AMOUNT);
447 break;
448 case ACT_CLOSE:
449 settings_persist();
450 request_screen(s_settings_return);
451 break;
452 case ACT_WIFI:
453 settings_persist();
454 set_wifi_progress("Scanning...", NULL);
455 request_screen(UI_SCREEN_WIFI_CONNECTING);
456 if (s_cb != NULL) { s_cb(UI_EVENT_WIFI_SCAN, 0); }
457 break;
458 case ACT_WIFI_CANCEL:
459 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(s_wifi_pass),
460 sizeof(s_wifi_pass));
461 request_screen(UI_SCREEN_AMOUNT);
462 break;
463 case ACT_WIFI_PASS_REVEAL:
464 /* Toggled in place, never via request_screen() — rebuilding the
465 * screen would discard what the operator has typed. */
466 if (s_wifi_pass_ta != NULL) {
467 bool masked = lv_textarea_get_password_mode(s_wifi_pass_ta);
468 lv_textarea_set_password_mode(s_wifi_pass_ta, !masked);
469 if (s_wifi_eye_lbl != NULL) {
470 /* Glyph shows what the next tap does. */
471 lv_label_set_text(s_wifi_eye_lbl,
472 masked ? LV_SYMBOL_EYE_CLOSE
473 : LV_SYMBOL_EYE_OPEN);
474 }
475 }
476 break;
477 case ACT_RESET:
478 open_reset_confirm(); /* ask before wiping */
479 break;
480 case ACT_RESET_CONFIRM:
481 /* Wipe stored settings (brightness, auto, Wi-Fi) and reboot — the
482 * device comes back up into first-run Wi-Fi setup. */
483 settings_factory_reset();
484 esp_restart();
485 break;
486 case ACT_RESET_CANCEL:
487 close_reset_confirm();
488 break;
489 }
490}

635/* user_data encodes which fee and direction: 0 max-, 1 max+, 2 prio-, 3 prio+. */
636static void fee_step_cb(lv_event_t *e) {
637 intptr_t code = reinterpret_cast<intptr_t>(lv_event_get_user_data(e));
638 uint32_t *v = (code < 2) ? &s_maxfee_gwei : &s_prio_gwei;
639 bool inc = (code & 1) != 0;
640
641 if (inc) {
642 *v = (*v + FEE_STEP_GWEI > FEE_MAX_GWEI) ? FEE_MAX_GWEI : *v + FEE_STEP_GWEI;
643 } else {
644 *v = (*v < FEE_MIN_GWEI + FEE_STEP_GWEI) ? FEE_MIN_GWEI : *v - FEE_STEP_GWEI;
645 }
646 /* The tip can never exceed the cap. */
647 if (s_prio_gwei > s_maxfee_gwei) { s_prio_gwei = s_maxfee_gwei; }
648 fee_update_labels();
649}

1195static uint32_t admin_penalty_ms(uint8_t fails) {
1196 if (fails < 3U) { return 0U; }
1197 uint32_t shift = static_cast<uint32_t>(fails) - 3U;
1198 if (shift > 6U) { shift = 6U; } /* cap before the shift overflows */
1199 uint32_t secs = 1U << shift;
1200 if (secs > 60U) { secs = 60U; }
1201 return secs * 1000U;
1202}

1204/* Seconds left on the penalty, 0 once it has elapsed. */
1205static uint32_t admin_lock_remaining_s(void) {
1206 if (s_admin_lock_ms == 0U) { return 0U; }
1207 const uint32_t elapsed = lv_tick_elaps(s_admin_lock_start);
1208 if (elapsed >= s_admin_lock_ms) {
1209 s_admin_lock_ms = 0U;
1210 return 0U;
1211 }
1212 return ((s_admin_lock_ms - elapsed) + 999U) / 1000U;
1213}

1225

1227 const char *txt = lv_textarea_get_text(s_admin_ta);
1228 strncpy(code, (txt != NULL) ? txt : "", sizeof(code) - 1);
1229 code[sizeof(code) - 1] = '\0';

1231 if (s_req_screen == UI_SCREEN_ADMIN_SET) {
1232 if (!s_admin_confirming) {

1235 (void)snprintf(msg, sizeof(msg), "At least %d digits",
1236 ADMIN_CODE_MIN);

1299 }
1300 lv_textarea_set_text(s_admin_ta, "");

1684/******************************************************************
1685 * 9. UI task — owns LVGL init and the handler loop
1686 ******************************************************************/
1687static void ui_task(void *arg) {
1688 (void)arg;
1689
1690 lv_init();
1691
1692 tft.init();
1693 tft.setRotation(0); /* portrait, 240x320 */
1694 tft.invertDisplay(true); /* CYD ILI9341 panel renders inverted otherwise */
1695
1696 /* CYD "milky gamma" fix: the 1-USB ILI9341_2 panels ship with a gamma
1697 * curve that crushes smooth gradients into visible bands — solid colours
1698 * and text look fine, but anti-aliased greys (e.g. the logo edges) come
1699 * out blocky. Re-select a built-in gamma curve via GAMMASET (0x26) to get
1700 * a clean ramp. See TFT_eSPI discussion #3018.
1701 */
1702 tft.writecommand(0x26); /* GAMMASET */
1703 tft.writedata(0x02);
1704 delay(120);
1705 tft.writecommand(0x26);
1706 tft.writedata(0x01);
1707
1708 /* White, not black: the backlight comes on below, before LVGL's first
1709 * frame, and the theme is white — black made power-on flash. */
1710 tft.fillScreen(TFT_WHITE);
1711
1712 touchSPI.begin(T_CLK, T_MISO, T_MOSI, T_CS);
1713 touch.begin(touchSPI);
1714 touch.setRotation(0); /* match the panel orientation */
1715
1716 /* Take over the backlight pin with LEDC PWM (after tft.init has touched
1717 * it) so brightness is dimmable from the settings menu. Restore the saved
1718 * level from NVS (defaults to 80% if never set). */
1719 s_brightness = settings_get_brightness();
1720 backlight_init(s_brightness);
1721
1722 lv_disp_draw_buf_init(&s_draw_buf, s_buf, NULL, SCR_W * 40);
1723 lv_disp_drv_init(&s_disp_drv);
1724 s_disp_drv.hor_res = SCR_W;
1725 s_disp_drv.ver_res = SCR_H;
1726 s_disp_drv.flush_cb = disp_flush;
1727 s_disp_drv.draw_buf = &s_draw_buf;
1728 lv_disp_drv_register(&s_disp_drv);
1729
1730 lv_indev_drv_init(&s_indev_drv);
1731 s_indev_drv.type = LV_INDEV_TYPE_POINTER;
1732 s_indev_drv.read_cb = indev_read;
1733 lv_indev_drv_register(&s_indev_drv);
1734
1735 const esp_timer_create_args_t targs = {
1736 .callback = &tick_cb,
1737 .arg = NULL,
1738 .dispatch_method = ESP_TIMER_TASK,
1739 .name = "lv_tick",
1740 .skip_unhandled_events = true,
1741 };
1742 esp_timer_handle_t th;
1743 if (esp_timer_create(&targs, &th) == ESP_OK) {
1744 (void)esp_timer_start_periodic(th, LV_TICK_PERIOD_MS * 1000);
1745 }
1746
1747 ESP_LOGI(TAG, "UI initialized (LVGL %d.%d + TFT_eSPI/XPT2046)",
1748 lv_version_major(), lv_version_minor());
1749
1750 while (true) {
1751 if (s_screen_dirty) {
1752 s_screen_dirty = false;
1753 render_requested_screen();
1754 }
1755 /* Boot-status update from the main task; targeted rather than a splash
1756 * rebuild, which would restart the logo on every step. */
1757 if (s_boot_step_dirty) {
1758 s_boot_step_dirty = false;
1759 if ((s_req_screen == UI_SCREEN_SPLASH) && (s_boot_step_lbl != NULL)) {
1760 lv_label_set_text(s_boot_step_lbl, s_boot_step);
1761 }
1762 }
1763 lv_timer_handler();
1764
1765 vTaskDelay(pdMS_TO_TICKS(5));
1766 }
1767}
