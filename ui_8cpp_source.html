126/******************************************************************
127 * Touch (calibrated raw range → screen coords; landscape rotation 1)
128 ******************************************************************/
129/* XPT2046_Touchscreen.setRotation(1) already swaps axes for landscape;
130 * we just map raw 200..3800 → screen 0..320 / 0..240 like esp32-loot does. */
131static bool poll_touch(int16_t *sx, int16_t *sy) {
132 if (!touch.tirqTouched() || !touch.touched()) return false;
133 TS_Point p = touch.getPoint();
134
135 int16_t mx = map(p.x, 200, 3800, 0, SCR_W);
136 int16_t my = map(p.y, 200, 3800, 0, SCR_H);
137 if (mx < 0) { mx = 0; }
138 if (mx >= SCR_W) { mx = SCR_W - 1; }
139 if (my < 0) { my = 0; }
140 if (my >= SCR_H) { my = SCR_H - 1; }
141
142 *sx = mx;
143 *sy = my;
144 return true;
145}

147/* Require a "finger up" moment between taps to suppress the chained-button
148 * cascade that happens when the user holds the screen or taps rapidly while
149 * the screen transitions (Cancel/CONFIRM/Send all sit at the same y). */
150static bool s_finger_lifted = true;
151
152static bool tap(int16_t *x, int16_t *y) {
153 bool pressed_now = touch.tirqTouched() && touch.touched();
154 if (!pressed_now) {
155 s_finger_lifted = true;
156 return false;
157 }
158 if (!s_finger_lifted) {
159 return false; /* sustained press — already handled */
160 }
161 uint32_t now = millis();
162 if (now - s_last_tap_ms < 150) {
163 return false;
164 }
165 if (!poll_touch(x, y)) {
166 return false;
167 }
168 s_last_tap_ms = now;
169 s_finger_lifted = false;
170 return true;
171}

320 bool can_retry = (s_tx_state == UI_TX_STATE_DONE) ||

335/******************************************************************
336 * Main UI task — drives touch + redraws
337 ******************************************************************/
338static void ui_task(void *arg) {
339 (void)arg;
340 while (true) {
341 if (s_redraw) {
342 switch (s_screen) {
343 case UI_SCREEN_SPLASH: draw_splash(); break;
344 case UI_SCREEN_AMOUNT: draw_amount_screen(); break;
345 case UI_SCREEN_CONFIRM: draw_confirm_screen(); break;
346 case UI_SCREEN_TX_STATUS: draw_tx_status_screen(); break;
347 }
348 s_redraw = false;
349 }
350
351 /* Incremental refresh of amount-entry value as user taps +/- */
352 if (s_screen == UI_SCREEN_AMOUNT && s_amount_units != s_last_drawn_amt) {
353 draw_amount_value();
354 }
355
356 int16_t x, y;
357 if (tap(&x, &y)) {
358 switch (s_screen) {
359 case UI_SCREEN_AMOUNT: handle_amount_tap(x, y); break;
360 case UI_SCREEN_CONFIRM: handle_confirm_tap(x, y); break;
361 case UI_SCREEN_TX_STATUS: handle_tx_status_tap(x, y); break;
362 default: break;
363 }
364 }
365
366 vTaskDelay(pdMS_TO_TICKS(20));
367 }
368}
