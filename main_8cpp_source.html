97static void parse_address(const char *hex, uint8_t out[20])
98{
99 const char *p = hex;
100 if ((p[0] == '0') && ((p[1] == 'x') || (p[1] == 'X'))) {
101 p += 2;
102 }
103 (void)memset(out, 0, 20U);
104 size_t i;
105 for (i = 0U; (i < 20U) && (p[0] != '\0') && (p[1] != '\0'); i++) {
106 uint8_t hi = (uint8_t)((*p >= 'a') ? (*p - 'a' + 10) :
107 (*p >= 'A') ? (*p - 'A' + 10) : (*p - '0'));
108 p++;
109 uint8_t lo = (uint8_t)((*p >= 'a') ? (*p - 'a' + 10) :
110 (*p >= 'A') ? (*p - 'A' + 10) : (*p - '0'));
111 p++;
112 out[i] = (uint8_t)((hi << 4U) | lo);
113 }
114}

116static void build_usdc_calldata(uint8_t out[68], const char *to_hex, uint64_t amount)
117{
118 (void)memset(out, 0, 68U);
119 (void)memcpy(out, TRANSFER_SELECTOR, 4U);
120 uint8_t addr[20];
121 parse_address(to_hex, addr);
122 (void)memcpy(out + 4U + 12U, addr, 20U);
123
124 size_t j;
125 for (j = 0U; j < 8U; j++) {
126 out[67U - j] = (uint8_t)((amount >> (8U * j)) & 0xFFU);
127 }
128}

130/******************************************************************
131 * Sign + broadcast for a given amount
132 *
133 * Returns true on success and writes the tx hash (with 0x prefix) to
134 * tx_hash_out. On failure, writes a short error message to err_out.
135 ******************************************************************/

137static bool sign_and_broadcast(CryptnoxWallet &wallet,
138 Pn532NfcTransport &transport,
139 uint64_t amount_units,
140 char *tx_hash_out, size_t tx_hash_max,
141 char *err_out, size_t err_max)
142{
143 uint8_t card_pin[CW_MAX_PIN_LENGTH];
144 (void)memset(card_pin, 0U, sizeof(card_pin));
145 (void)memcpy(card_pin, CARD_PIN,
146 (CARD_PIN_LEN < CW_MAX_PIN_LENGTH) ? CARD_PIN_LEN : CW_MAX_PIN_LENGTH);
147
148 uint8_t calldata[68];
149 build_usdc_calldata(calldata, "0x" ADDR_TO, amount_units);
150
151 uint64_t nonce = 0U;
152 if (!eth_rpc_get_nonce(&nonce)) {
153 (void)snprintf(err_out, err_max, "RPC: get nonce failed");
154 return false;
155 }
156
157 eth_tx_t tx;
158 (void)memset(&tx, 0, sizeof(tx));
159 tx.chain_id = CHAIN_ID_SEPOLIA;
160 tx.nonce = nonce;
161 tx.max_priority_fee = MAX_PRIORITY_FEE;
162 tx.max_fee = MAX_FEE;
163 tx.gas_limit = GAS_LIMIT_ERC20;
164 tx.eth_value = 0U;
165 tx.calldata = calldata;
166 tx.calldata_len = sizeof(calldata);
167 parse_address("0x" ADDR_USDC, tx.to);
168
169 uint8_t unsigned_tx[TX_BUF_SIZE];
170 size_t unsigned_len = eth_rlp_encode_unsigned(&tx, unsigned_tx, sizeof(unsigned_tx));
171 if (unsigned_len == 0U) {
172 (void)snprintf(err_out, err_max, "RLP encode overflow");
173 return false;
174 }
175
176 uint8_t hash[CW_HASH_SIZE];
177 keccak256(unsigned_tx, unsigned_len, hash);
178
179 if (!s_user_cancelled) {
180 ui_show_tx_status(UI_TX_STATE_PLACE_CARD, "Hold card to reader");
181 }
182
183 /* Manual connect loop with cancel checks between PN532 polls — replaces
184 * wallet.connect() so a Cancel from the user aborts within one PN532
185 * timeout (~5 s) instead of the full retry budget (~26 s). */
186 CW_SecureSession session;
187 bool connected = false;
188 for (int attempt = 0; (attempt < CW_CONNECT_MAX_ATTEMPTS) && !connected; attempt++) {
189 if (s_user_cancelled) {
190 return false;
191 }
192 if (transport.inListPassiveTarget()) {
193 vTaskDelay(pdMS_TO_TICKS(200));
194 if (wallet.establishSecureChannel(session)) {
195 connected = true;
196 break;
197 }
198 }
199 vTaskDelay(pdMS_TO_TICKS(200));
200 }
201 if (s_user_cancelled) {
202 return false;
203 }
204 if (!connected) {
205 (void)snprintf(err_out, err_max, "Card not found");
206 return false;
207 }
208
209 if (!s_user_cancelled) {
210 ui_show_tx_status(UI_TX_STATE_SIGNING, NULL);
211 }
212
213 /* BIP32 Ethereum derivation path: m/44'/60'/0'/0/0 */
214 static const uint8_t eth_path[20] = {
215 0x80U, 0x00U, 0x00U, 0x2CU,
216 0x80U, 0x00U, 0x00U, 0x3CU,
217 0x80U, 0x00U, 0x00U, 0x00U,
218 0x00U, 0x00U, 0x00U, 0x00U,
219 0x00U, 0x00U, 0x00U, 0x00U,
220 };
221
222 CW_SignRequest req(session,
223 CW_SIGN_DERIVE_K1,
224 CW_SIGN_SIG_ECDSA_LOW_S,
225 CW_SIGN_WITH_PIN);
226 req.hash = hash;
227 req.hashLength = static_cast<uint8_t>(CW_HASH_SIZE);
228 req.derivePath = eth_path;
229 req.derivePathLength = static_cast<uint8_t>(sizeof(eth_path));
230 (void)memcpy(req.pin, card_pin, CW_MAX_PIN_LENGTH);
231
232 CW_SignResult result = wallet.sign(req);
233 wallet.disconnect(session);
234
235 if (result.errorCode != CW_OK) {
236 (void)snprintf(err_out, err_max, "Sign error 0x%02X",
237 static_cast<unsigned int>(result.errorCode));
238 return false;
239 }
240
241 const uint8_t *sig_r = result.signature + CW_SIG_R_OFFSET;
242 const uint8_t *sig_s = result.signature + CW_SIG_S_OFFSET;
243
244 if (!s_user_cancelled) {
245 ui_show_tx_status(UI_TX_STATE_SENDING, NULL);
246 }
247
248 uint8_t v = eth_rpc_ecrecover_parity(hash, sig_r, sig_s);
249
250 uint8_t signed_tx[TX_BUF_SIZE];
251 size_t signed_len = eth_rlp_encode_signed(&tx, v, sig_r, sig_s,
252 signed_tx, sizeof(signed_tx));
253 if (signed_len == 0U) {
254 (void)snprintf(err_out, err_max, "RLP signed overflow");
255 return false;
256 }
257
258 if (!eth_rpc_send_raw_tx(signed_tx, signed_len, tx_hash_out, tx_hash_max)) {
259 (void)snprintf(err_out, err_max, "Broadcast failed");
260 return false;
261 }
262
263 return true;
264}

345 uint64_t pending_amount = 0U;

347

379 } else if (ok) {
380 ui_show_tx_status(UI_TX_STATE_DONE, tx_hash);
381 } else {

383 }
384 break;
385 }

388 ui_show_amount_entry();
389 break;
390
391 default:
392 break;
393 }
394 }
