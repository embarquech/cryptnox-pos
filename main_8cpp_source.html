155static bool build_usdc_calldata(uint8_t out[68], const char *to_hex, uint64_t amount)
156{
157 (void)memset(out, 0, 68U);
158 (void)CW_Utils::safe_memcpy(out, 68U, TRANSFER_SELECTOR, 4U);
159 uint8_t addr[20];
160 if (!eth_addr_parse(to_hex, addr)) {
161 return false;
162 }
163 (void)CW_Utils::safe_memcpy(out + 4U + 12U, 68U - (4U + 12U), addr, 20U);
164
165 size_t j;
166 for (j = 0U; j < 8U; j++) {
167 out[67U - j] = static_cast<uint8_t>((amount >> (8U * j)) & 0xFFU);
168 }
169 return true;
170}

218static bool sign_and_broadcast(CryptnoxWallet &wallet,
219 Pn532NfcTransport &transport,
220 uint64_t amount_units,
221 const char *pin, size_t pin_chars,
222 char *tx_hash_out, size_t tx_hash_max,
223 char *err_out, size_t err_max)
224{
225 uint8_t calldata[68];
226 if (!build_usdc_calldata(calldata, "0x" ADDR_TO, amount_units)) {
227 (void)snprintf(err_out, err_max, "Bad ADDR_TO in config");
228 return false;
229 }
230
231 uint64_t nonce = 0U;
232 if (!eth_rpc_get_nonce(&nonce)) {
233 (void)snprintf(err_out, err_max, "RPC: get nonce failed");
234 return false;
235 }
236
237 eth_tx_t tx;
238 (void)memset(&tx, 0, sizeof(tx));
239 tx.chain_id = CHAIN_ID_SEPOLIA;
240 tx.nonce = nonce;
241 /* Fees come from the settings menu (defaulting to the config.h values on
242 * first boot); config.h still owns the gas limit. The user edits Gwei, so
243 * scale to wei. Keep the tip <= the cap or the tx is malformed. */
244 uint64_t max_fee_wei = (uint64_t)settings_get_max_fee_gwei() * 1000000000ULL;
245 uint64_t prio_fee_wei = (uint64_t)settings_get_priority_fee_gwei() * 1000000000ULL;
246 if (prio_fee_wei > max_fee_wei) { prio_fee_wei = max_fee_wei; }
247 tx.max_priority_fee = prio_fee_wei;
248 tx.max_fee = max_fee_wei;
249 tx.gas_limit = GAS_LIMIT_ERC20;
250 tx.eth_value = 0U;
251 tx.calldata = calldata;
252 tx.calldata_len = sizeof(calldata);
253 if (!eth_addr_parse("0x" ADDR_USDC, tx.to)) {
254 (void)snprintf(err_out, err_max, "Bad ADDR_USDC in config");
255 return false;
256 }
257
258 uint8_t unsigned_tx[TX_BUF_SIZE];
259 size_t unsigned_len = eth_rlp_encode_unsigned(&tx, unsigned_tx, sizeof(unsigned_tx));
260 if (unsigned_len == 0U) {
261 (void)snprintf(err_out, err_max, "RLP encode overflow");
262 return false;
263 }
264
265 uint8_t hash[CW_HASH_SIZE];
266 keccak256(unsigned_tx, unsigned_len, hash);
267
268 /* From here the message hash and (soon) the signature live on the stack.
269 * Scrub them and the encoded transactions on every exit path below. */
270 WipeGuard g_unsigned(unsigned_tx, sizeof(unsigned_tx));
271 WipeGuard g_hash(hash, sizeof(hash));
272
273 if (!s_user_cancelled) {
274 ui_show_tx_status(UI_TX_STATE_PLACE_CARD, "Hold card to reader");
275 }
276
277 /* Start from a clean reader state: a previous attempt that ended with the
278 * card ripped away mid-exchange can leave a stale target selected, which
279 * would make every InListPassiveTarget below time out. */
280 transport.resetReader();
281
282 /* Manual connect loop with cancel checks between PN532 polls — replaces
283 * wallet.connect() so a Cancel from the user aborts within one PN532
284 * timeout. Give the user up to 60 s to present the card. */
285 const int64_t card_wait_us = 60LL * 1000000LL; /* 60 seconds */
286 const int64_t start_us = esp_timer_get_time();
287 CW_SecureSession session;
288 bool connected = false;
289 while (!connected) {
290 if (s_user_cancelled) {
291 return false;
292 }
293 if ((esp_timer_get_time() - start_us) > card_wait_us) {
294 break; /* timed out waiting for the card */
295 }
296 if (transport.inListPassiveTarget()) {
297 /* Card tapped — show immediate feedback while the secure channel
298 * comes up. */
299 ui_show_tx_status(UI_TX_STATE_PROCESSING, NULL);
300 vTaskDelay(pdMS_TO_TICKS(200));
301 if (wallet.establishSecureChannel(session)) {
302 connected = true;
303 break;
304 }
305 /* Card pulled away (or channel failed) — the PN532 still holds the
306 * now-dead target selected, which makes every following
307 * InListPassiveTarget time out. Release it so the reader can see
308 * the card again, then back to the tap prompt. */
309 transport.resetReader();
310 if (!s_user_cancelled) {
311 ui_show_tx_status(UI_TX_STATE_PLACE_CARD, "Hold card to reader");
312 }
313 }
314 vTaskDelay(pdMS_TO_TICKS(200));
315 }
316 if (s_user_cancelled) {
317 return false;
318 }
319 if (!connected) {
320 (void)snprintf(err_out, err_max, "Card not found");
321 return false;
322 }
323
324 if (!s_user_cancelled) {
325 ui_show_tx_status(UI_TX_STATE_SIGNING, NULL);
326 }
327
328 /* BIP32 Ethereum derivation path: m/44'/60'/0'/0/0 */
329 static const uint8_t eth_path[20] = {
330 0x80U, 0x00U, 0x00U, 0x2CU,
331 0x80U, 0x00U, 0x00U, 0x3CU,
332 0x80U, 0x00U, 0x00U, 0x00U,
333 0x00U, 0x00U, 0x00U, 0x00U,
334 0x00U, 0x00U, 0x00U, 0x00U,
335 };
336
337 CW_SignRequest req(session,
338 CW_SIGN_DERIVE_K1,
339 CW_SIGN_SIG_ECDSA_LOW_S,
340 CW_SIGN_WITH_PIN);
341 req.hash = hash;
342 req.hashLength = static_cast<uint8_t>(CW_HASH_SIZE);
343 req.derivePath = eth_path;
344 req.derivePathLength = static_cast<uint8_t>(sizeof(eth_path));
345
346 /* Copy the operator-entered PIN into the request as late as possible;
347 * req.pin is zero-initialised by the CW_SignRequest constructor. */
348 const size_t copy_len = (pin_chars < CW_MAX_PIN_LENGTH) ? pin_chars
349 : CW_MAX_PIN_LENGTH;
350 (void)CW_Utils::safe_memcpy(req.pin, sizeof(req.pin),
351 reinterpret_cast<const uint8_t *>(pin),
352 copy_len);
353
354 CW_SignResult result = wallet.sign(req);
355 WipeGuard g_sig(result.signature, sizeof(result.signature));
356 wallet.disconnect(session);
357
358 /* scrub the PIN immediately after use (secure_wipe is not
359 * dead-store-eliminated); ~CW_SignRequest wipes it again as a backstop. */
360 CW_Utils::secure_wipe(req.pin, sizeof(req.pin));
361
362 if (result.errorCode != CW_OK) {
363 (void)snprintf(err_out, err_max, "Sign error 0x%02X",
364 static_cast<unsigned int>(result.errorCode));
365 return false;
366 }
367
368 const uint8_t *sig_r = result.signature + CW_SIG_R_OFFSET;
369 const uint8_t *sig_s = result.signature + CW_SIG_S_OFFSET;
370
371 if (!s_user_cancelled) {
372 ui_show_tx_status(UI_TX_STATE_SENDING, NULL);
373 }
374
375 /* the parity recovery now fails explicitly instead of silently
376 * defaulting to v=0, so the operator sees the actual root cause. */
377 uint8_t v = 0U;
378 switch (eth_rpc_ecrecover_parity(hash, sig_r, sig_s, &v)) {
379 case ETH_RPC_PARITY_OK:
380 break;
381 case ETH_RPC_PARITY_MISMATCH:
382 (void)snprintf(err_out, err_max, "Card-address mismatch");
383 return false;
384 case ETH_RPC_PARITY_RPC_ERROR:
385 default:
386 (void)snprintf(err_out, err_max, "RPC error (parity)");
387 return false;
388 }
389
390 uint8_t signed_tx[TX_BUF_SIZE];
391 WipeGuard g_signed(signed_tx, sizeof(signed_tx));
392 size_t signed_len = eth_rlp_encode_signed(&tx, v, sig_r, sig_s,
393 signed_tx, sizeof(signed_tx));
394 if (signed_len == 0U) {
395 (void)snprintf(err_out, err_max, "RLP signed overflow");
396 return false;
397 }
398
399 /* last cancel check right before the irreversible broadcast. */
400 if (s_user_cancelled) {
401 return false;
402 }
403
404 if (!eth_rpc_send_raw_tx(signed_tx, signed_len, tx_hash_out, tx_hash_max)) {
405 (void)snprintf(err_out, err_max, "Broadcast failed");
406 return false;
407 }
408
409 return true;
410}

432static void ensure_wifi(void)
433{
434 char ssid[33] = { 0 };
435 char pass[65] = { 0 };
436 if (settings_get_wifi(ssid, sizeof(ssid), pass, sizeof(pass))) {
437 bool ok = net_wifi_connect(ssid, pass);
438 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(pass), sizeof(pass));
439 if (ok) { return; }
440 }
441
442 /* No usable saved network — forced interactive setup. */
443 net_wifi_ap_t aps[16];
444 uint16_t n = net_wifi_scan(aps, 16);
445 ui_show_wifi_list(aps, n);
446
447 ui_msg_t msg;
448 while (true) {
449 if (xQueueReceive(s_ui_queue, &msg, portMAX_DELAY) != pdTRUE) { continue; }
450
451 if (msg.event == UI_EVENT_WIFI_TRY) {
452 char w_ssid[33] = { 0 };
453 char w_pass[65] = { 0 };
454 bool ok = false;
455 if (ui_take_wifi_creds(w_ssid, sizeof(w_ssid),
456 w_pass, sizeof(w_pass)) > 0U) {
457 ui_show_wifi_connecting(w_ssid);
458 ok = net_wifi_connect(w_ssid, w_pass);
459 if (ok) { settings_set_wifi(w_ssid, w_pass); }
460 }
461 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(w_pass), sizeof(w_pass));
462 if (ok) { return; }
463 }
464
465 /* WIFI_SCAN, a failed connect, or any stray event: re-scan and show
466 * the list again so the user stays in setup until connected. */
467 n = net_wifi_scan(aps, 16);
468 ui_show_wifi_list(aps, n);
469 }

472extern "C" void app_main(void)
473{
474 ESP_LOGI(TAG, "===== cryptnox-pos boot =====");
475#ifdef CRYPTNOX_POS_DEV_BUILD
476 /* build timestamp helps firmware fingerprinting — dev builds only. */
477 ESP_LOGI(TAG, "Build: %s %s", __DATE__, __TIME__);
478#endif
479 ESP_LOGI(TAG, "Tag: %s", APP_VERSION_TAG);
480
481 /* The SDK's per-APDU PN532 log flood was trimmed upstream (SDK main);
482 * INFO now only emits a few init lines worth keeping ("I2C wake-up
483 * trigger", "PN532 initialized"). Keep the adapters at WARN — their
484 * per-APDU chatter is still verbose. */
485 esp_log_level_set("pn532", ESP_LOG_INFO);
486 esp_log_level_set("pn532_adapter", ESP_LOG_WARN);
487 esp_log_level_set("Pn532NfcTransport", ESP_LOG_WARN);
488
489 /* PN532 I2C busy-polling: the chip NACKs its own address while busy —
490 * that is the normal PN532 I2C protocol, but IDF's i2c.master driver
491 * logs a 4-line error burst for every poll. Mute the tag; real I2C
492 * failures still propagate through the SDK's return codes. */
493 esp_log_level_set("i2c.master", ESP_LOG_NONE);
494
495 /* ── NVS first: required by the WiFi driver AND by the UI task, which
496 * reads the saved backlight level / Wi-Fi credentials at startup. ── */
497 esp_err_t nvs_ret = nvs_flash_init();
498 if ((nvs_ret == ESP_ERR_NVS_NO_FREE_PAGES) ||
499 (nvs_ret == ESP_ERR_NVS_NEW_VERSION_FOUND)) {
500 ESP_ERROR_CHECK(nvs_flash_erase());
501 nvs_ret = nvs_flash_init();
502 }
503 ESP_ERROR_CHECK(nvs_ret);
504
505 /* ── UI: splash visible while the rest boots ───────── */
506 s_ui_queue = xQueueCreate(8, sizeof(ui_msg_t));
507 ui_init(ui_event_dispatch);
508 ui_set_addresses("0x" ADDR_USDC, "0x" ADDR_TO);
509 ui_show_splash();
510
511 /* ── PN532 NFC reader ──────────────────────────────────────── */
512 pn532_t nfc;
513 (void)memset(&nfc, 0, sizeof(nfc));
514
515 pn532_config_t nfc_cfg;
516 (void)memset(&nfc_cfg, 0, sizeof(nfc_cfg));
517 nfc_cfg.transport = PN532_TRANSPORT_I2C;
518 nfc_cfg.i2c_port = PN532_I2C_PORT;
519 nfc_cfg.pin_sda = PN532_SDA;
520 nfc_cfg.pin_scl = PN532_SCL;
521 nfc_cfg.pin_irq = PN532_IRQ;
522 nfc_cfg.pin_rst = PN532_RST;
523 nfc_cfg.i2c_clock_hz = PN532_I2C_HZ;
524
525 if (pn532_init(&nfc, &nfc_cfg) != ESP_OK) {
526 ESP_LOGE(TAG, "PN532 init failed");
527 ui_show_tx_status(UI_TX_STATE_FAILED, "PN532 not found");
528 return;
529 }
530
531 /* ── Wallet ────────────────────────────────────────────────── */
532 NullLogger logger;
533 (void)logger.begin(115200UL);
534 ESP32CryptoProvider cryptoProvider;
535 Pn532NfcTransport nfcTransport(&nfc, logger);
536 ESP32Platform platform;
537 CryptnoxWallet wallet(nfcTransport, logger, cryptoProvider, platform);
538
539 if (!wallet.begin()) {
540 ESP_LOGE(TAG, "Wallet begin failed");
541 ui_show_tx_status(UI_TX_STATE_FAILED, "Wallet init failed");
542 return;
543 }
544
545 /* ── WiFi + RPC ────────────────────────────────────────────── */
546 eth_rpc_init(RPC_URL, "0x" ADDR_FROM);
547#if defined(RPC_PROJECT_ID) && defined(RPC_API_SECRET)
548 eth_rpc_set_auth(RPC_PROJECT_ID, RPC_API_SECRET);
549#endif
550#ifdef RPC_CA_CERT_PEM
551 /* pin the RPC endpoint's certificate instead of the CA bundle. */
552 eth_rpc_set_ca_cert(RPC_CA_CERT_PEM);
553#endif
554 net_wifi_init();
555 /* Connect with saved creds, or run the first-run picker until connected
556 * (config.h Wi-Fi is no longer used). */
557 ensure_wifi();
558
559 /* block on a first SNTP sync so TLS certificate validity-period
560 * checks run against real time instead of the 1970 epoch. Retry a couple
561 * of rounds — flaky uplinks (phone hotspots) often need a second try. */
562 bool time_ok = false;
563 for (int attempt = 0; (attempt < 3) && !time_ok; attempt++) {
564 time_ok = net_time_sync(15000U);
565 }
566 if (!time_ok) {
567 ESP_LOGE(TAG, "SNTP time sync failed");
568 ui_show_tx_status(UI_TX_STATE_FAILED, "Time sync failed - check network");
569 return;
570 }
571
572 /* One RPC round-trip at boot, for two reasons at once: it proves the
573 * endpoint is reachable, and it is the first request whose authenticated
574 * Date header can contradict the unauthenticated SNTP clock we just set.
575 * Without it a clock wrong in the *forward* direction — the direction that
576 * carries a certificate past its notAfter — would sail through boot and
577 * only surface mid-payment, with a customer waiting.
578 *
579 * Retried like the sync above, since a single failure is more often a
580 * flaky uplink than a hostile one. The payment path re-checks every
581 * request regardless, so this is early warning, not the enforcement. */
582 bool rpc_ok = false;
583 for (int attempt = 0; (attempt < 3) && !rpc_ok; attempt++) {
584 uint64_t boot_nonce = 0U;
585 rpc_ok = eth_rpc_get_nonce(&boot_nonce);
586 }
587 if (!rpc_ok) {
588 ESP_LOGE(TAG, "RPC unreachable or clock rejected at boot");
589 ui_show_tx_status(UI_TX_STATE_FAILED, "RPC/clock check failed - see log");
590 return;
591 }
592
593 ESP_LOGI(TAG, "Ready");
594
595 /* ── Main interaction loop ────────────────────────────────── */
596 ui_show_amount_entry();
597
598 uint64_t pending_amount = 0U;
599 ui_msg_t msg;
600
601 while (true) {
602 if (xQueueReceive(s_ui_queue, &msg, portMAX_DELAY) != pdTRUE) {
603 continue;
604 }
605
606 switch (msg.event) {
607 case UI_EVENT_AMOUNT_CONFIRMED:
608 pending_amount = msg.payload;
609 ui_show_confirm(pending_amount, "0x" ADDR_TO);
610 break;
611
612 case UI_EVENT_CONFIRM_CANCEL:
613 ui_show_amount_entry();
614 break;
615
616 case UI_EVENT_PIN_ENTERED: {
617 if (pending_amount == 0U) {
618 /* No fresh AMOUNT_CONFIRMED preceded this — likely a stale
619 * event from a stuck touch or queue replay. Drop it. */
620 ESP_LOGW(TAG, "stale PIN_ENTERED ignored");
621 break;
622 }
623 /* Fetch the keypad PIN (UI wipes its own copy on read). */
624 char pin[16] = { 0 };
625 size_t pin_chars = ui_take_pin(pin, sizeof(pin));
626
627 s_user_cancelled = false;
628 char tx_hash[68] = { 0 };
629 char err_msg[64] = { 0 };
630 bool ok = sign_and_broadcast(wallet, nfcTransport, pending_amount,
631 pin, pin_chars,
632 tx_hash, sizeof(tx_hash),
633 err_msg, sizeof(err_msg));
634 /* scrub our copy of the PIN as soon as signing is done. */
635 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(pin), sizeof(pin));
636
637 pending_amount = 0U; /* next sign requires fresh New Payment flow */
638 if (s_user_cancelled) {
639 /* Cancel during PLACE_CARD — UI already on amount entry. */
640 } else if (ok) {
641 /* Broadcast accepted only means "entered the mempool" — a
642 * POS must not claim Approved until the tx is mined with
643 * status 0x1. Poll the receipt (Sepolia block ~12 s). */
644 ui_show_tx_status(UI_TX_STATE_CONFIRMING, NULL);
645 eth_rpc_receipt_result_t rc = ETH_RPC_RECEIPT_PENDING;
646 const int64_t deadline =
647 esp_timer_get_time() + 120LL * 1000000LL; /* 120 s */
648 while (esp_timer_get_time() < deadline) {
649 rc = eth_rpc_get_tx_receipt(tx_hash);
650 if ((rc == ETH_RPC_RECEIPT_SUCCESS) ||
651 (rc == ETH_RPC_RECEIPT_REVERTED)) {
652 break;
653 }
654 /* PENDING or transient RPC error — try again. */
655 vTaskDelay(pdMS_TO_TICKS(4000));
656 }
657 if (rc == ETH_RPC_RECEIPT_SUCCESS) {
658 ESP_LOGI(TAG, "Tx confirmed on-chain");
659 ui_show_tx_status(UI_TX_STATE_DONE, tx_hash);
660 } else if (rc == ETH_RPC_RECEIPT_REVERTED) {
661 ESP_LOGE(TAG, "Tx reverted on-chain: %s", tx_hash);
662 ui_show_tx_status(UI_TX_STATE_FAILED, "Payment reverted");
663 } else {
664 ESP_LOGE(TAG, "Tx not confirmed after 120 s: %s", tx_hash);
665 ui_show_tx_status(UI_TX_STATE_FAILED,
666 "Confirmation timeout - check explorer");
667 }
668 } else {
669 ui_show_tx_status(UI_TX_STATE_FAILED, err_msg);
670 }
671 break;
672 }
673
674 case UI_EVENT_TX_RETRY:
675 ui_show_amount_entry();
676 break;
677
678 case UI_EVENT_WIFI_SCAN: {
679 /* Scan runs in this task so the UI stays responsive. */
