97/* ── USDC ERC-20 transfer(address,uint256) selector + calldata ── */
98static const uint8_t TRANSFER_SELECTOR[4] = { 0xa9U, 0x05U, 0x9cU, 0xbbU };
99#define ABI_SELECTOR_LEN 4U /* transfer(address,uint256) selector */
100#define ABI_WORD_LEN 32U /* one ABI-encoded argument word */
101#define USDC_CALLDATA_LEN (ABI_SELECTOR_LEN + (2U * ABI_WORD_LEN)) /* 68 */
102#define ABI_TO_OFFSET (ABI_SELECTOR_LEN + (ABI_WORD_LEN - ETH_ADDR_LEN))

166static void build_usdc_calldata(uint8_t out[USDC_CALLDATA_LEN],
167 const uint8_t to[ETH_ADDR_LEN],
168 uint64_t amount)
169{
170 CW_Utils::secure_wipe(out, USDC_CALLDATA_LEN);
171 (void)CW_Utils::safe_memcpy(out, USDC_CALLDATA_LEN,
172 TRANSFER_SELECTOR, ABI_SELECTOR_LEN);
173 (void)CW_Utils::safe_memcpy(out + ABI_TO_OFFSET,
174 USDC_CALLDATA_LEN - ABI_TO_OFFSET,
175 to, ETH_ADDR_LEN);
176
177 size_t j;
178 for (j = 0U; j < sizeof(amount); j++) {
179 out[(USDC_CALLDATA_LEN - 1U) - j] =
180 static_cast<uint8_t>((amount >> (8U * j)) & 0xFFU);
181 }
182}

230static bool sign_and_broadcast(CryptnoxWallet &wallet,
231 Pn532NfcTransport &transport,
232 const pos_amount_t *amount,
233 const pos_addr_t *to,
234 const char *pin, size_t pin_chars,
235 char *tx_hash_out, size_t tx_hash_max,
236 char *err_out, size_t err_max)
237{
238 /* Reconcile amount + recipient right before they enter the calldata
239 * (§3.2/§7.1); a mismatch means the working copy was corrupted. */
240 if (!IS_TRUE32(amount_consistent(amount)) ||
241 !IS_TRUE32(address_consistent(to))) {
242 pos_handle_anomaly("pre-calldata reconcile");
243 (void)snprintf(err_out, err_max, "Integrity check failed");
244 return false;
245 }
246 const uint64_t amount_units = amount->amount_minor;
247
248 uint8_t calldata[USDC_CALLDATA_LEN];
249 build_usdc_calldata(calldata, to->addr, amount_units);
250
251 uint64_t nonce = 0U;
252 if (!eth_rpc_get_nonce(&nonce)) {
253 (void)snprintf(err_out, err_max, "RPC: get nonce failed");
254 return false;
255 }
256
257 eth_tx_t tx;
258 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(&tx), sizeof(tx));
259 tx.chain_id = CHAIN_ID_SEPOLIA;
260 tx.nonce = nonce;
261 /* Fees come from the settings menu (defaulting to the config.h values on
262 * first boot); config.h still owns the gas limit. The user edits Gwei, so
263 * scale to wei. Keep the tip <= the cap or the tx is malformed. */
264 uint64_t max_fee_wei = (uint64_t)settings_get_max_fee_gwei() * 1000000000ULL;
265 uint64_t prio_fee_wei = (uint64_t)settings_get_priority_fee_gwei() * 1000000000ULL;
266 if (prio_fee_wei > max_fee_wei) { prio_fee_wei = max_fee_wei; }
267 tx.max_priority_fee = prio_fee_wei;
268 tx.max_fee = max_fee_wei;
269 tx.gas_limit = GAS_LIMIT_ERC20;
270 tx.eth_value = 0U;
271 tx.calldata = calldata;
272 tx.calldata_len = sizeof(calldata);
273 if (!eth_addr_parse("0x" ADDR_USDC, tx.to)) {
274 (void)snprintf(err_out, err_max, "Bad ADDR_USDC in config");
275 return false;
276 }
277
278 uint8_t unsigned_tx[TX_BUF_SIZE];
279 size_t unsigned_len = eth_rlp_encode_unsigned(&tx, unsigned_tx, sizeof(unsigned_tx));
280 if (unsigned_len == 0U) {
281 (void)snprintf(err_out, err_max, "RLP encode overflow");
282 return false;
283 }
284
285 uint8_t hash[CW_HASH_SIZE];
286 keccak256(unsigned_tx, unsigned_len, hash);
287
288 /* From here the message hash and (soon) the signature live on the stack.
289 * Scrub them and the encoded transactions on every exit path below. */
290 WipeGuard g_unsigned(unsigned_tx, sizeof(unsigned_tx));
291 WipeGuard g_hash(hash, sizeof(hash));
292
293 if (!s_user_cancelled) {
294 ui_show_tx_status(UI_TX_STATE_PLACE_CARD, "Hold card to reader");
295 }
296
297 /* Start from a clean reader state: a previous attempt that ended with the
298 * card ripped away mid-exchange can leave a stale target selected, which
299 * would make every InListPassiveTarget below time out. */
300 transport.resetReader();
301
302 /* Manual connect loop with cancel checks between PN532 polls — replaces
303 * wallet.connect() so a Cancel from the user aborts within one PN532
304 * timeout. Give the user up to 60 s to present the card. */
305 const int64_t card_wait_us = 60LL * 1000000LL; /* 60 seconds */
306 const int64_t start_us = esp_timer_get_time();
307 CW_SecureSession session;
308 bool connected = false;
309 while (!connected) {
310 if (s_user_cancelled) {
311 return false;
312 }
313 if ((esp_timer_get_time() - start_us) > card_wait_us) {
314 break; /* timed out waiting for the card */
315 }
316 if (transport.inListPassiveTarget()) {
317 /* Card tapped — show immediate feedback while the secure channel
318 * comes up. */
319 ui_show_tx_status(UI_TX_STATE_PROCESSING, NULL);
320 vTaskDelay(pdMS_TO_TICKS(200));
321 if (wallet.establishSecureChannel(session)) {
322 connected = true;
323 break;
324 }
325 /* Card pulled away (or channel failed) — the PN532 still holds the
326 * now-dead target selected, which makes every following
327 * InListPassiveTarget time out. Release it so the reader can see
328 * the card again, then back to the tap prompt. */
329 transport.resetReader();
330 if (!s_user_cancelled) {
331 ui_show_tx_status(UI_TX_STATE_PLACE_CARD, "Hold card to reader");
332 }
333 }
334 vTaskDelay(pdMS_TO_TICKS(200));
335 }
336 if (s_user_cancelled) {
337 return false;
338 }
339 if (!connected) {
340 (void)snprintf(err_out, err_max, "Card not found");
341 return false;
342 }
343
344 if (!s_user_cancelled) {
345 ui_show_tx_status(UI_TX_STATE_SIGNING, NULL);
346 }
347
348 /* BIP32 Ethereum derivation path: m/44'/60'/0'/0/0 */
349 static const uint8_t eth_path[20] = {
350 0x80U, 0x00U, 0x00U, 0x2CU,
351 0x80U, 0x00U, 0x00U, 0x3CU,
352 0x80U, 0x00U, 0x00U, 0x00U,
353 0x00U, 0x00U, 0x00U, 0x00U,
354 0x00U, 0x00U, 0x00U, 0x00U,
355 };
356
357 CW_SignRequest req(session,
358 CW_SIGN_DERIVE_K1,
359 CW_SIGN_SIG_ECDSA_LOW_S,
360 CW_SIGN_WITH_PIN);
361 req.hash = hash;
362 req.hashLength = static_cast<uint8_t>(CW_HASH_SIZE);
363 req.derivePath = eth_path;
364 req.derivePathLength = static_cast<uint8_t>(sizeof(eth_path));
365
366 /* Re-reconcile amount + recipient right before signing — this is the last
367 * point before the card produces an irreversible signature over the
368 * calldata (§3.2/§7.1). */
369 if (!IS_TRUE32(amount_consistent(amount)) ||
370 !IS_TRUE32(address_consistent(to))) {
371 pos_handle_anomaly("pre-sign reconcile");
372 (void)snprintf(err_out, err_max, "Integrity check failed");
373 return false;
374 }
375
376 /* Copy the operator-entered PIN into the request as late as possible;
377 * req.pin is zero-initialised by the CW_SignRequest constructor. */
378 const size_t copy_len = (pin_chars < CW_MAX_PIN_LENGTH) ? pin_chars
379 : CW_MAX_PIN_LENGTH;
380 (void)CW_Utils::safe_memcpy(req.pin, sizeof(req.pin),
381 reinterpret_cast<const uint8_t *>(pin),
382 copy_len);
383
384 CW_SignResult result = wallet.sign(req);
385 WipeGuard g_sig(result.signature, sizeof(result.signature));
386 wallet.disconnect(session);
387
388 /* scrub the PIN immediately after use (secure_wipe is not
389 * dead-store-eliminated); ~CW_SignRequest wipes it again as a backstop. */
390 CW_Utils::secure_wipe(req.pin, sizeof(req.pin));
391
392 if (result.errorCode != CW_OK) {
393 (void)snprintf(err_out, err_max, "Sign error 0x%02X",
394 static_cast<unsigned int>(result.errorCode));
395 return false;
396 }
397
398 const uint8_t *sig_r = result.signature + CW_SIG_R_OFFSET;
399 const uint8_t *sig_s = result.signature + CW_SIG_S_OFFSET;
400
401 if (!s_user_cancelled) {
402 ui_show_tx_status(UI_TX_STATE_SENDING, NULL);
403 }
404
405 /* the parity recovery now fails explicitly instead of silently
406 * defaulting to v=0, so the operator sees the actual root cause. */
407 uint8_t v = 0U;
408 switch (eth_rpc_ecrecover_parity(hash, sig_r, sig_s, &v)) {
409 case ETH_RPC_PARITY_OK:
410 break;
411 case ETH_RPC_PARITY_MISMATCH:
412 (void)snprintf(err_out, err_max, "Card-address mismatch");
413 return false;
414 case ETH_RPC_PARITY_RPC_ERROR:
415 default:
416 (void)snprintf(err_out, err_max, "RPC error (parity)");
417 return false;
418 }
419
420 uint8_t signed_tx[TX_BUF_SIZE];
421 WipeGuard g_signed(signed_tx, sizeof(signed_tx));
422 size_t signed_len = eth_rlp_encode_signed(&tx, v, sig_r, sig_s,
423 signed_tx, sizeof(signed_tx));
424 if (signed_len == 0U) {
425 (void)snprintf(err_out, err_max, "RLP signed overflow");
426 return false;
427 }
428
429 /* last cancel check right before the irreversible broadcast. */
430 if (s_user_cancelled) {
431 return false;
432 }
433
434 if (!eth_rpc_send_raw_tx(signed_tx, signed_len, tx_hash_out, tx_hash_max)) {
435 (void)snprintf(err_out, err_max, "Broadcast failed");
436 return false;
437 }
438
439 return true;
440}

455/* Startup retry budgets. The effort is spent out here rather than inside
456 * net_wifi_connect(), since each call resets the association properly:
457 * 3 x (1 + WIFI_MAX_RETRY) associations, 45 s worst case. */
458#define WIFI_SAVED_ATTEMPTS 3U
459#define TIME_SYNC_ATTEMPTS 3U
460
461/* Picker notes, shared by the boot bring-up and the settings Wi-Fi change so the
462 * two cannot drift apart. */
463static const char *const NOTE_JOIN_FAILED =
464 "Could not join that network - check the password";
465static const char *const NOTE_NO_TIME =
466 "No network time - this Wi-Fi has no usable internet";
467
468/* Credentials the picker just joined with, held until a clock sync proves the
469 * network usable end to end (see wifi_keep_or_drop). Associating is not enough:
470 * a captive-portal or offline AP joins fine and would then be reached for on
471 * every boot. Deferring the write also means a transient NTP outage never
472 * erases a saved network that does work. */
473static char s_join_ssid[33] = { 0 };
474static char s_join_pass[65] = { 0 };
475
482static void wifi_keep_or_drop(bool keep)
483{
484 if (keep && (s_join_ssid[0] != '\0')) {
485 settings_set_wifi(s_join_ssid, s_join_pass);
486 ESP_LOGI(TAG, "saved network '%s' (clock synced)", s_join_ssid);
487 }
488 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(s_join_pass), sizeof(s_join_pass));
489 (void)memset(s_join_ssid, 0, sizeof(s_join_ssid));
490}

529static bool ensure_wifi(bool try_saved, const char *note)
530{
531 char ssid[33] = { 0 };
532 char pass[65] = { 0 };
533 if (try_saved &&
534 settings_get_wifi(ssid, sizeof(ssid), pass, sizeof(pass))) {
535 /* Unattended: report on the splash rather than flashing up a setup
536 * screen the operator never asked for. */
537 ui_set_boot_status("Connecting to Wi-Fi");
538 bool ok = false;
539 for (uint32_t a = 1U; (a <= WIFI_SAVED_ATTEMPTS) && !ok; a++) {
540 ESP_LOGI(TAG, "Wi-Fi '%s': attempt %" PRIu32 "/%u",
541 ssid, a, WIFI_SAVED_ATTEMPTS);
542 ok = net_wifi_connect(ssid, pass);
543 }
544 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(pass), sizeof(pass));
545 if (ok) { return false; } /* splash never left the screen */
546 ESP_LOGW(TAG, "saved network '%s' failed %u times - opening the picker",
547 ssid, WIFI_SAVED_ATTEMPTS);
548 note = "Could not join the saved network";
549 }
550
551 /* No usable saved network — forced interactive setup. */
552 net_wifi_ap_t aps[16];
553 uint16_t n = net_wifi_scan(aps, 16);
554 ui_show_wifi_list(aps, n, note);
555
556 ui_msg_t msg;
557 while (true) {
558 if (xQueueReceive(s_ui_queue, &msg, portMAX_DELAY) != pdTRUE) { continue; }
559
560 if (msg.event == UI_EVENT_WIFI_TRY) {
561 char w_ssid[33] = { 0 };
562 char w_pass[65] = { 0 };
563 bool ok = false;
564 if (ui_take_wifi_creds(w_ssid, sizeof(w_ssid),
565 w_pass, sizeof(w_pass)) > 0U) {
566 /* Interactive: the operator expects to see the attempt. */
567 ui_show_wifi_connecting(w_ssid);
568 ok = net_wifi_connect(w_ssid, w_pass);
569 if (ok) {
570 /* Staged, not saved — wifi_keep_or_drop() decides once the
571 * clock has proven this network carries real internet. */
572 (void)snprintf(s_join_ssid, sizeof(s_join_ssid), "%s", w_ssid);
573 (void)snprintf(s_join_pass, sizeof(s_join_pass), "%s", w_pass);
574 }
575 }
576 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(w_pass), sizeof(w_pass));
577 if (ok) { return true; } /* the picker owns the screen */
578 note = NOTE_JOIN_FAILED;
579 } else if (msg.event == UI_EVENT_WIFI_SCAN) {
580 note = NULL; /* rescan asked for — the old reason is stale */
581 }
582
583 /* WIFI_SCAN, a failed connect, or any stray event: re-scan and show
584 * the list again so the user stays in setup until connected. */
585 n = net_wifi_scan(aps, 16);
586 ui_show_wifi_list(aps, n, note);
587 }
588}

597static bool sync_time(void)
598{
599 for (uint32_t a = 1U; a <= TIME_SYNC_ATTEMPTS; a++) {
600 ESP_LOGI(TAG, "SNTP sync: attempt %" PRIu32 "/%u", a, TIME_SYNC_ATTEMPTS);

606extern "C" void app_main(void)
607{
608 ESP_LOGI(TAG, "===== cryptnox-pos boot =====");
609#ifdef CRYPTNOX_POS_DEV_BUILD
610 /* build timestamp helps firmware fingerprinting — dev builds only. */
611 ESP_LOGI(TAG, "Build: %s %s", __DATE__, __TIME__);
612#endif
613 ESP_LOGI(TAG, "Tag: %s", APP_VERSION_TAG);
614
615 /* The SDK's per-APDU PN532 log flood was trimmed upstream (SDK main);
616 * INFO now only emits a few init lines worth keeping ("I2C wake-up
617 * trigger", "PN532 initialized"). Keep the adapters at WARN — their
618 * per-APDU chatter is still verbose. */
619 esp_log_level_set("pn532", ESP_LOG_INFO);
620 esp_log_level_set("pn532_adapter", ESP_LOG_WARN);
621 esp_log_level_set("Pn532NfcTransport", ESP_LOG_WARN);
622
623 /* PN532 I2C busy-polling: the chip NACKs its own address while busy —
624 * that is the normal PN532 I2C protocol, but IDF's i2c.master driver
625 * logs a 4-line error burst for every poll. Mute the tag; real I2C
626 * failures still propagate through the SDK's return codes. */
627 esp_log_level_set("i2c.master", ESP_LOG_NONE);
628
629 /* ── NVS first: required by the WiFi driver AND by the UI task, which
630 * reads the saved backlight level / Wi-Fi credentials at startup. ── */
631 esp_err_t nvs_ret = nvs_flash_init();
632 if ((nvs_ret == ESP_ERR_NVS_NO_FREE_PAGES) ||
633 (nvs_ret == ESP_ERR_NVS_NEW_VERSION_FOUND)) {
634 ESP_ERROR_CHECK(nvs_flash_erase());
635 nvs_ret = nvs_flash_init();
636 }
637 ESP_ERROR_CHECK(nvs_ret);
638
639 /* ── UI: splash visible while the rest boots ───────── */
640 s_ui_queue = xQueueCreate(8, sizeof(ui_msg_t));
641 ui_init(ui_event_dispatch);
642 ui_set_addresses("0x" ADDR_USDC, "0x" ADDR_TO);
643 /* No ui_show_splash() here — ui_init() already selects it, and asking twice
644 * races the UI task into rebuilding the screen and replaying the logo. */
645
646 /* Parse ADDR_TO twice into the dual store (§7.1). Also runs the EIP-55
647 * checksum on ADDR_TO once at boot — a mistyped recipient is caught here,
648 * before any payment. */
649 if (!eth_addr_parse("0x" ADDR_TO, s_dest.addr) ||
650 !eth_addr_parse("0x" ADDR_TO, s_dest.addr_echo)) {
651 ESP_LOGE(TAG, "Bad ADDR_TO in config");
652 ui_show_tx_status(UI_TX_STATE_FAILED, "Bad ADDR_TO in config");
653 return;
654 }
655 /* Warn if ADDR_TO carries no EIP-55 checksum (no upper-case hex letter) —
656 * the boot-time typo check above is a no-op on an all-lowercase address.
657 * Manual scan, not strpbrk: ADDR_TO is a literal, so strpbrk(...)==NULL
658 * folds to a provably-false pointer compare (-Werror=address). */
659 bool addr_checksummed = false;
660 for (const char *pc = ADDR_TO; *pc != '\0'; ++pc) {
661 if ((*pc >= 'A') && (*pc <= 'F')) { addr_checksummed = true; break; }
662 }
663 if (!addr_checksummed) {
664 ESP_LOGW(TAG, "ADDR_TO is all-lowercase: no EIP-55 checksum verified");
665 }
666
667 /* ── PN532 NFC reader ──────────────────────────────────────── */
668 ui_set_boot_status("Starting NFC reader");
669 pn532_t nfc;
670 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(&nfc), sizeof(nfc));
671
672 pn532_config_t nfc_cfg;
673 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(&nfc_cfg), sizeof(nfc_cfg));
674 nfc_cfg.transport = PN532_TRANSPORT_I2C;
675 nfc_cfg.i2c_port = PN532_I2C_PORT;
676 nfc_cfg.pin_sda = PN532_SDA;
677 nfc_cfg.pin_scl = PN532_SCL;
678 nfc_cfg.pin_irq = PN532_IRQ;
679 nfc_cfg.pin_rst = PN532_RST;
680 nfc_cfg.i2c_clock_hz = PN532_I2C_HZ;
681
682 /* Keep the error code — unplugged reader and misconfigured bus look
683 * identical on screen otherwise. */
684 esp_err_t nfc_ret = pn532_init(&nfc, &nfc_cfg);
685 if (nfc_ret != ESP_OK) {
686 ESP_LOGE(TAG, "PN532 bus init failed: %s", esp_err_to_name(nfc_ret));
687 ui_show_boot_error(UI_BOOT_ERR_NFC, esp_err_to_name(nfc_ret));
688 return;
689 }
690
691 /* pn532_init() only brings up the bus and ignores its own probe results
692 * (pn532.h), so it returns ESP_OK with no reader attached. Probe here —
693 * 0 means no answer — or an absent reader is reported as a wallet fault. */
694 uint32_t nfc_fw = pn532_get_firmware_version(&nfc);
695 if (nfc_fw == 0U) {
696 ESP_LOGE(TAG, "PN532 did not answer GetFirmwareVersion - reader absent?");
697 ui_show_boot_error(UI_BOOT_ERR_NFC, "No answer to GetFirmwareVersion");
698 return;
699 }
700 ESP_LOGI(TAG, "PN532 firmware: IC 0x%02X, version %u.%u",
701 (unsigned)((nfc_fw >> 24) & 0xFFU),
702 (unsigned)((nfc_fw >> 16) & 0xFFU),
703 (unsigned)((nfc_fw >> 8) & 0xFFU));
704
705 /* ── Wallet ────────────────────────────────────────────────── */
706 ui_set_boot_status("Opening wallet");
707 NullLogger logger;
708 (void)logger.begin(115200UL);
709 ESP32CryptoProvider cryptoProvider;
710 Pn532NfcTransport nfcTransport(&nfc, logger);
711 ESP32Platform platform;
712 CryptnoxWallet wallet(nfcTransport, logger, cryptoProvider, platform);
713
714 if (!wallet.begin()) {
715 ESP_LOGE(TAG, "Wallet begin failed");
716 ui_show_boot_error(UI_BOOT_ERR_WALLET, NULL);
717 return;
718 }
719
720 /* ── WiFi + RPC ────────────────────────────────────────────── */
721 eth_rpc_init(RPC_URL, "0x" ADDR_FROM);
722#if defined(RPC_PROJECT_ID) && defined(RPC_API_SECRET)
723 eth_rpc_set_auth(RPC_PROJECT_ID, RPC_API_SECRET);
724#endif
725#ifdef RPC_CA_CERT_PEM
726 /* pin the RPC endpoint's certificate instead of the CA bundle. */
727 eth_rpc_set_ca_cert(RPC_CA_CERT_PEM);
728#endif
729 /* ── First run: greet, then set up ────────────────────────── */
730 /* A missing admin code means a virgin or factory-reset terminal, since the
731 * reset erases it too. Greet before the setup steps start asking for a
732 * network and a code — it is the one moment we have the operator's attention
733 * and nothing to demand of them yet. */
734 const bool first_run = !settings_has_admin_code();
735 if (first_run) {
736 ui_show_welcome();
737 wait_for_ui_event(UI_EVENT_WELCOME_DONE);
738
739 /* The code comes BEFORE the network, and not for tidiness: the Wi-Fi
740 * picker carries a back arrow that the UI task honours on its own, which
741 * drops the operator on the amount screen — burger included — while main
742 * is still blocked here. With no code stored yet, that burger opened the
743 * settings freely. Creating the code first closes that window; the
744 * creation screen itself has no way out. */
745 ESP_LOGI(TAG, "no admin code - first-run setup");
746 ui_show_admin_set();
747 wait_for_ui_event(UI_EVENT_ADMIN_SET);
748 }
749
750 ui_set_boot_status("Starting network");
751 net_wifi_init();
752
753 /* Wi-Fi and a valid clock are one bring-up step, since TLS needs both: a
754 * failed sync sends the operator back to the picker with the reason rather
755 * than stranding the terminal on an error screen. */
756 bool try_saved = true;
757 const char *net_note = NULL;
758 while (true) {
759 if (ensure_wifi(try_saved, net_note)) {
760 ui_show_splash(); /* only when the picker took the screen */
761 }
762 ui_set_boot_status("Syncing clock");
763 if (sync_time()) {
764 wifi_keep_or_drop(true); /* proven usable — safe to persist */
765 break;
766 }
767 ESP_LOGE(TAG, "SNTP time sync failed on this network");
768 /* Drop the staged credentials, but leave an already-saved network alone:
769 * it may well work again after a reboot, and erasing it would cost the
770 * operator the password for what is often a transient outage. */
771 wifi_keep_or_drop(false);
772 /* Force the picker: retrying the same network loops straight back here. */
773 try_saved = false;
774 net_note = NOTE_NO_TIME;
775 }
776
777 /* One RPC round-trip at boot, for two reasons at once: it proves the
778 * endpoint is reachable, and it is the first request whose authenticated
779 * Date header can contradict the unauthenticated SNTP clock we just set.
780 * Without it a clock wrong in the *forward* direction — the direction that
781 * carries a certificate past its notAfter — would sail through boot and
782 * only surface mid-payment, with a customer waiting.
783 *
784 * Retried like the sync above, since a single failure is more often a
785 * flaky uplink than a hostile one. The payment path re-checks every
786 * request regardless, so this is early warning, not the enforcement. */
787 bool rpc_ok = false;
788 for (int attempt = 0; (attempt < 3) && !rpc_ok; attempt++) {
789 uint64_t boot_nonce = 0U;
790 rpc_ok = eth_rpc_get_nonce(&boot_nonce);
791 }
792 if (!rpc_ok) {
793 ESP_LOGE(TAG, "RPC unreachable or clock rejected at boot");
794 ui_show_tx_status(UI_TX_STATE_FAILED, "RPC/clock check failed - see log");
795 return;
796 }
797
798 ESP_LOGI(TAG, "Ready");
799
800 /* ── Main interaction loop ────────────────────────────────── */
801 ui_show_amount_entry();
802
803 pos_amount_t pending_amount;
804 pos_amount_set(&pending_amount, 0U);
805 ui_msg_t msg;
806
807 while (true) {
808 if (xQueueReceive(s_ui_queue, &msg, portMAX_DELAY) != pdTRUE) {
809 continue;
810 }
811
812 switch (msg.event) {
813 case UI_EVENT_AMOUNT_CONFIRMED:
814 pos_amount_set(&pending_amount, msg.payload);
815 /* Reconcile amount + recipient before they are shown to the
816 * customer — displayed value must equal what gets signed. */
817 if (!IS_TRUE32(amount_consistent(&pending_amount)) ||
818 !IS_TRUE32(address_consistent(&s_dest))) {
819 pos_handle_anomaly("pre-display reconcile");
820 ui_show_tx_status(UI_TX_STATE_FAILED, "Integrity check failed");
821 pos_amount_set(&pending_amount, 0U);
822 break;
823 }
824 ui_show_confirm(pending_amount.amount_minor, "0x" ADDR_TO);
825 break;
826
827 case UI_EVENT_CONFIRM_CANCEL:
828 ui_show_amount_entry();
829 break;
830
831 case UI_EVENT_PIN_ENTERED: {
832 if (pending_amount.amount_minor == 0U) {
833 /* No fresh AMOUNT_CONFIRMED preceded this — likely a stale
834 * event from a stuck touch or queue replay. Drop it. */
835 ESP_LOGW(TAG, "stale PIN_ENTERED ignored");
836 break;
837 }
838 /* Fetch the keypad PIN (UI wipes its own copy on read). */
839 char pin[16] = { 0 };
840 size_t pin_chars = ui_take_pin(pin, sizeof(pin));
841
842 s_user_cancelled = false;
843 char tx_hash[68] = { 0 };
844 char err_msg[64] = { 0 };
845 bool ok = sign_and_broadcast(wallet, nfcTransport,
846 &pending_amount, &s_dest,
847 pin, pin_chars,
848 tx_hash, sizeof(tx_hash),
849 err_msg, sizeof(err_msg));
850 /* scrub our copy of the PIN as soon as signing is done. */
851 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(pin), sizeof(pin));
852
853 /* Snapshot the decided amount for the final gate, then clear
854 * pending so the next sign needs a fresh New Payment flow. */
855 pos_amount_t decided = pending_amount;
856 pos_amount_set(&pending_amount, 0U);
857 if (s_user_cancelled) {
858 /* Cancel during PLACE_CARD — UI already on amount entry. */
859 } else if (ok) {
860 /* Broadcast accepted only means "entered the mempool" — a
861 * POS must not claim Approved until the tx is mined with
862 * status 0x1. Poll the receipt (Sepolia block ~12 s). */
863 ui_show_tx_status(UI_TX_STATE_CONFIRMING, NULL);
864 eth_rpc_receipt_result_t rc = ETH_RPC_RECEIPT_PENDING;
865 const int64_t deadline =
866 esp_timer_get_time() + 120LL * 1000000LL; /* 120 s */
867 while (esp_timer_get_time() < deadline) {
868 rc = eth_rpc_get_tx_receipt(tx_hash);
869 if ((rc == ETH_RPC_RECEIPT_SUCCESS) ||
870 (rc == ETH_RPC_RECEIPT_REVERTED)) {
871 break;
872 }
873 /* PENDING or transient RPC error — try again. */
874 vTaskDelay(pdMS_TO_TICKS(4000));
875 }
876 /* §4: render PAID only if the monotonic gate holds — the
877 * on-chain APPROVED verdict AND amount/recipient still
878 * self-consistent through the decide→render window. */
879 pos_verdict_t verdict = (rc == ETH_RPC_RECEIPT_SUCCESS)
880 ? POS_VERDICT_APPROVED
881 : POS_VERDICT_DECLINED;
882 bool32 decision =
883 run_payment_decision(&decided, &s_dest, verdict);
884 if (rc == ETH_RPC_RECEIPT_SUCCESS && IS_TRUE32(decision)) {
885 ESP_LOGI(TAG, "Tx confirmed on-chain");
886 ui_show_tx_status(UI_TX_STATE_DONE, tx_hash);
887 } else if (rc == ETH_RPC_RECEIPT_SUCCESS) {
888 /* Mined OK but the integrity gate failed — never show
889 * PAID on a corrupted decision. */
890 pos_handle_anomaly("final decision gate");
891 ESP_LOGE(TAG, "Integrity gate failed post-receipt: %s", tx_hash);
892 ui_show_tx_status(UI_TX_STATE_FAILED, "Integrity check failed");
893 } else if (rc == ETH_RPC_RECEIPT_REVERTED) {
894 ESP_LOGE(TAG, "Tx reverted on-chain: %s", tx_hash);
895 ui_show_tx_status(UI_TX_STATE_FAILED, "Payment reverted");
896 } else {
897 ESP_LOGE(TAG, "Tx not confirmed after 120 s: %s", tx_hash);
898 ui_show_tx_status(UI_TX_STATE_FAILED,
899 "Confirmation timeout - check explorer");
900 }
901 } else {
902 ui_show_tx_status(UI_TX_STATE_FAILED, err_msg);
903 }
904 break;
905 }
906
907 case UI_EVENT_TX_RETRY:
908 ui_show_amount_entry();
909 break;
910
911 case UI_EVENT_WIFI_SCAN: {
912 /* Scan runs in this task so the UI stays responsive. */
913 net_wifi_ap_t aps[16];
914 uint16_t n = net_wifi_scan(aps, 16);
915 /* Opened from settings, not after a failure — no note. */
916 ui_show_wifi_list(aps, n, NULL);
917 break;
918 }
919
920 case UI_EVENT_WIFI_TRY: {
921 char w_ssid[33] = { 0 };
922 char w_pass[65] = { 0 };
923 if (ui_take_wifi_creds(w_ssid, sizeof(w_ssid),
924 w_pass, sizeof(w_pass)) > 0U) {
925 ui_show_wifi_connecting(w_ssid);
926
927 /* Same rule as boot: associating proves nothing, so make the
928 * clock prove the uplink before overwriting saved credentials
929 * that may well be working. One round only — the operator is
930 * standing there and can tap again, where boot has to be
931 * patient on its own. */
932 const char *why = NULL;
933 if (!net_wifi_connect(w_ssid, w_pass)) {
934 why = NOTE_JOIN_FAILED;
935 } else if (!net_time_sync(15000U)) {
936 ESP_LOGW(TAG, "'%s' joined but has no network time -"
937 " not saved", w_ssid);
938 why = NOTE_NO_TIME;
939 } else {
940 settings_set_wifi(w_ssid, w_pass); /* persist for next boot */
941 ui_show_amount_entry();
942 }
943
944 if (why != NULL) {
945 /* Refusing to persist is not enough: the radio is still
946 * associated with the network we just rejected, and the
947 * picker's back arrow goes straight to amount entry. So
948 * without this the terminal looks ready while every
949 * payment fails at the RPC call. Roll back to the saved
950 * network — which boot already proved usable — before
951 * handing over the screen. Costs up to one association
952 * timeout, hence the progress screen. */
953 char b_ssid[33] = { 0 };
954 char b_pass[65] = { 0 };
955 if (settings_get_wifi(b_ssid, sizeof(b_ssid),
956 b_pass, sizeof(b_pass)) &&
957 (b_ssid[0] != '\0')) {
958 ESP_LOGW(TAG, "rolling back to saved network '%s'",
959 b_ssid);
960 ui_show_wifi_connecting(b_ssid);
961 (void)net_wifi_connect(b_ssid, b_pass);
962 }
963 CW_Utils::secure_wipe(
964 reinterpret_cast<uint8_t *>(b_pass), sizeof(b_pass));
965
966 /* Back to the picker with the reason, so another network
967 * can be chosen instead of a dead end. */
968 net_wifi_ap_t aps[16];
969 uint16_t n = net_wifi_scan(aps, 16);
970 ui_show_wifi_list(aps, n, why);
971 }
972 }
973 CW_Utils::secure_wipe(reinterpret_cast<uint8_t *>(w_pass), sizeof(w_pass));
974 break;
975 }
976
977 default:
978 break;
979 }
980 }
